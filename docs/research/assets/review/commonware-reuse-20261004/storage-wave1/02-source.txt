0: //! Shared validation for QMDB batch chains.
1: //!
2: //! A batch chain is a linked sequence of in-memory batches built on top of a DB state. Each batch
3: //! records its state via [`Bounds`] (where its operations sit in the log, and the root at each
4: //! applicable boundary) and the inactivity floor declared by its commit. Older batches in the chain
5: //! are tracked as [`AncestorBounds`] in newest-first order. Some may already be applied to the
6: //! database while others may not.
7: //!
8: //! Before applying a batch to the DB, the internal validation checks two things shared across QMDB
9: //! variants (any, immutable, keyless):
10: //!
11: //! - The batch is not stale: the current DB state must match either the batch's recorded DB state
12: //!   or one of its ancestor states.
13: //! - Commit floors are monotonically non-decreasing along the chain, and no floor exceeds
14: //!   its own commit location. Ancestors already applied to the database are skipped because their
15: //!   floors were validated when they were first applied. The rest of the chain and the tip are
16: //!   checked.
17: //!
18: //! Internal helpers walk the chain via weak parent references and snapshot ancestor bounds into a
19: //! `Vec` for storage on a merkleized batch.
20: 
21: use crate::{
22:     merkle::{Family, Location},
23:     qmdb::Error,
24: };
25: use commonware_cryptography::Digest;
26: use core::iter;
27: use std::sync::{Arc, Weak};

29: /// Identifies a QMDB state by its operation `size` and authenticated `root`.
30: #[derive(Clone, Copy, Debug)]
31: pub struct Commitment<F: Family, D: Digest> {
32:     /// Number of operations in the state.
33:     pub size: Location<F>,
34:     /// Root committing to those operations.
35:     pub root: D,
36: }
37: 
38: impl<F: Family, D: Digest> Commitment<F, D> {
39:     /// Create a [`Commitment`] from an operation `size` and its committing `root` digest.
40:     pub(crate) const fn new(size: Location<F>, root: D) -> Self {
41:         Self { size, root }
42:     }
43: }
44: 
45: // `Family` is not `PartialEq`, so deriving would demand `F: PartialEq` at every call site.
46: // Compare the fields directly, which needs only `Location<F>: PartialEq` and `D: Eq`.
47: impl<F: Family, D: Digest> PartialEq for Commitment<F, D> {
48:     fn eq(&self, other: &Self) -> bool {
49:         self.size == other.size && self.root == other.root
50:     }
51: }
52: 
53: impl<F: Family, D: Digest> Eq for Commitment<F, D> {}

55: /// Bounds declared by an ancestor batch's commit.
56: #[derive(Clone)]
57: pub struct AncestorBounds<F: Family, D: Digest> {
58:     /// Inactivity floor declared by the ancestor commit.
59:     pub floor: Location<F>,
60:     /// [`Commitment`] after the ancestor batch.
61:     pub state: Commitment<F, D>,
62: }
63: 
64: /// Position and inactivity-floor state for a merkleized QMDB batch.
65: #[derive(Clone)]
66: pub struct Bounds<F: Family, D: Digest> {
67:     /// [`Commitment`] immediately before this batch's own operations.
68:     pub base: Commitment<F, D>,
69:     /// [`Commitment`] at the boundary between applied database operations and operations kept in
70:     /// this batch chain.
71:     ///
72:     /// Usually this is the database state when the batch chain was created.
73:     pub db: Commitment<F, D>,
74:     /// This batch's tip [`Commitment`]: the state after all its operations.
75:     pub tip: Commitment<F, D>,
76:     /// Ancestor bounds in newest-first order.
77:     pub ancestors: Vec<AncestorBounds<F, D>>,
78:     /// Inactivity floor declared by this batch's commit.
79:     pub inactivity_floor: Location<F>,
80: }
81: 
82: impl<F: Family, D: Digest> Bounds<F, D> {
83:     /// Create initial bounds for a batch built directly from the current database state.
84:     ///
85:     /// The base, DB boundary, and tip all coincide at `state`, with no ancestors.
86:     pub(crate) const fn from_db(state: Commitment<F, D>, inactivity_floor: Location<F>) -> Self {
87:         Self {
88:             base: state,
89:             db: state,
90:             tip: state,
91:             ancestors: Vec::new(),
92:             inactivity_floor,
93:         }
94:     }
95: 
96:     /// Validate that this batch can be applied to the current database state.
97:     pub(crate) fn validate_apply_to(
98:         &self,
99:         current: Commitment<F, D>,
100:         current_floor: Location<F>,
101:     ) -> Result<(), Error<F>> {
102:         validate_batch_applicable(current, self.db, &self.ancestors)?;
103:         validate_commit_floors(
104:             current_floor,
105:             current.size,
106:             &self.ancestors,
107:             self.inactivity_floor,
108:             self.tip
109:                 .size
110:                 .checked_sub(1)
111:                 .expect("merkleized batch includes a commit"),
112:         )
113:     }
114: }

116: /// Iterate over a batch's live ancestors, starting at `parent`.
117: ///
118: /// Iteration stops when a weak parent reference cannot be upgraded.
119: pub(crate) fn ancestors<T, P>(
120:     parent: Option<Weak<T>>,
121:     mut parent_of: P,
122: ) -> impl Iterator<Item = Arc<T>>
123: where
124:     P: for<'a> FnMut(&'a T) -> Option<&'a Weak<T>>,
125: {
126:     let mut next = parent.as_ref().and_then(Weak::upgrade);
127:     iter::from_fn(move || {
128:         let batch = next.take()?;
129:         next = parent_of(&batch).and_then(Weak::upgrade);
130:         Some(batch)
131:     })
132: }
133: 
134: /// Iterate over a strong parent followed by its live ancestors.
135: pub(crate) fn parent_and_ancestors<T, P, I>(
136:     parent: Option<&Arc<T>>,
137:     mut ancestors_of: P,
138: ) -> impl Iterator<Item = Arc<T>> + use<T, P, I>
139: where
140:     P: FnMut(&Arc<T>) -> I,
141:     I: IntoIterator<Item = Arc<T>>,
142: {
143:     parent.cloned().into_iter().flat_map(move |parent| {
144:         let ancestors = ancestors_of(&parent);
145:         iter::once(parent).chain(ancestors)
146:     })
147: }
148: 
149: /// Collect ancestor bounds in newest-first order.
150: pub(crate) fn collect_ancestor_bounds<T, F, D, I, L, C>(
151:     ancestors: I,
152:     floor: L,
153:     state: C,
154: ) -> Vec<AncestorBounds<F, D>>
155: where
156:     F: Family,
157:     D: Digest,
158:     I: IntoIterator<Item = Arc<T>>,
159:     L: Fn(&T) -> Location<F>,
160:     C: Fn(&T) -> Commitment<F, D>,
161: {
162:     ancestors
163:         .into_iter()
164:         .map(|batch| AncestorBounds {
165:             floor: floor(&batch),
166:             state: state(&batch),
167:         })
168:         .collect()
169: }

171: /// Advance the inherited DB boundary past applied ancestors no longer reachable
172: /// through the weak parent chain.
173: pub(crate) fn effective_boundary<F: Family, D: Digest>(
174:     inherited: Commitment<F, D>,
175:     oldest_live_base: Option<Commitment<F, D>>,
176: ) -> Commitment<F, D> {
177:     oldest_live_base
178:         .filter(|base| base.size > inherited.size)
179:         .unwrap_or(inherited)
180: }

182: /// Validate that a batch can be applied to the database at the given [`Commitment`].
183: ///
184: /// A batch is applicable if the database has not advanced since the batch was created, if all
185: /// ancestors are already applied, or if the database has advanced to one of the batch's ancestor
186: /// [`Commitment`]s.
187: pub(crate) fn validate_batch_applicable<F: Family, D: Digest>(
188:     current: Commitment<F, D>,
189:     batch_db: Commitment<F, D>,
190:     ancestors: &[AncestorBounds<F, D>],
191: ) -> Result<(), Error<F>> {
192:     // A separate base check is unnecessary: a direct batch's base is `batch_db`, while a child
193:     // batch's base is its first ancestor.
194:     if current == batch_db || ancestors.iter().any(|ancestor| ancestor.state == current) {
195:         return Ok(());
196:     }
197: 
198:     Err(Error::StaleBatch)
199: }
200: 
201: /// Validate commit-floor monotonicity for a batch chain.
202: ///
203: /// Ancestors are stored newest-first. Validation walks them in reverse so unapplied ancestors are
204: /// checked oldest-to-newest, then checks the tip. Ancestors at or below `db_size` are already
205: /// applied locally and are skipped.
206: pub(crate) fn validate_commit_floors<F: Family, D: Digest>(
207:     starting_floor: Location<F>,
208:     db_size: Location<F>,
209:     ancestors: &[AncestorBounds<F, D>],
210:     tip_floor: Location<F>,
211:     tip_commit_loc: Location<F>,
212: ) -> Result<(), Error<F>> {
213:     let mut prev_floor = starting_floor;
214:     for ancestor in ancestors.iter().rev() {
215:         if ancestor.state.size <= db_size {
216:             continue;
217:         }
218: 
219:         let ancestor_commit_loc = ancestor.state.size - 1;
220:         if ancestor.floor < prev_floor {
221:             return Err(Error::FloorRegressed(ancestor.floor, prev_floor));
222:         }
223:         if ancestor.floor > ancestor_commit_loc {
224:             return Err(Error::FloorBeyondSize(ancestor.floor, ancestor_commit_loc));
225:         }
226:         prev_floor = ancestor.floor;
227:     }
228: 
229:     if tip_floor < prev_floor {
230:         return Err(Error::FloorRegressed(tip_floor, prev_floor));
231:     }
232:     if tip_floor > tip_commit_loc {
233:         return Err(Error::FloorBeyondSize(tip_floor, tip_commit_loc));
234:     }
235:     Ok(())
236: }
