9: /// Target state to sync to.
10: ///
11: /// `PartialEq`, `Eq`, and `Clone` are implemented manually to avoid requiring `F` to implement
12: /// them.
13: #[derive(Debug)]
14: pub struct Target<F: Family, D: Digest> {
15:     /// The ops root the sync engine verifies streaming batches against.
16:     pub root: D,
17:     /// Range of operations to sync
18:     pub range: NonEmptyRange<Location<F>>,
19: }
20: 
21: impl<F: Family, D: Digest> Target<F, D> {
22:     /// Create a sync target.
23:     pub const fn new(root: D, range: NonEmptyRange<Location<F>>) -> Self {
24:         Self { root, range }
25:     }
26: 
27:     /// Whether this target advances relative to `from`.
28:     ///
29:     /// Both targets are assumed to describe valid states of the same append-only QMDB. Because
30:     /// the root commits to the database size, valid targets at different sizes have distinct
31:     /// roots.
32:     pub fn advances(&self, from: &Self) -> bool {
33:         self.range.end().is_valid()
34:             && self.range.end() > from.range.end()
35:             && self.range.start() >= from.range.start()
36:     }
37: }

104: pub struct CompactTarget<F: Family, D: Digest> {
105:     /// Target database root.
106:     pub root: D,
107:     /// Target database size.
108:     pub size: Location<F>,
109: }

111: impl<F: Family, D: Digest> TryFrom<&CompactTarget<F, D>> for Target<F, D> {
112:     type Error = EngineError<F, D>;
113: 
114:     fn try_from(target: &CompactTarget<F, D>) -> Result<Self, Self::Error> {
115:         let end = target.size;
116:         let start =
117:             end.checked_sub(1)
118:                 .filter(|_| end.is_valid())
119:                 .ok_or(EngineError::InvalidTarget {
120:                     lower_bound_pos: Location::new(0),
121:                     upper_bound_pos: end,
122:                 })?;
123:         Ok(Self {
124:             root: target.root,
125:             range: non_empty_range!(start, end),
126:         })
127:     }
128: }

269:     fn test_advances() {
270:         let current_root = sha256::Digest::from([0; 32]);
271:         let advanced_root = sha256::Digest::from([1; 32]);
272:         let current = target(current_root, 10, 100);
273: 
274:         // End strictly increases, start does not decrease.
275:         assert!(target(advanced_root, 10, 101).advances(&current));
276:         assert!(target(advanced_root, 50, 200).advances(&current));
277: 
278:         // Same or smaller end does not advance.
279:         assert!(!target(current_root, 10, 100).advances(&current));
280:         assert!(!target(current_root, 10, 50).advances(&current));
281: 
282:         // A start moving backward does not advance, even with a larger end.
283:         assert!(!target(advanced_root, 5, 200).advances(&current));
284: 
285:         // An end outside the location domain does not advance.
286:         let beyond = target(advanced_root, 10, *MmrFamily::MAX_LEAVES + 1);
287:         assert!(!beyond.advances(&current));
288:     }
