229: impl<F, E, C, I, H, U, const N: usize, S> CurrentStaged<F, E, C, I, H, U, N, S>
230: where
231:     F: Graftable,
232:     E: Context,
233:     U: Update,
234:     C: Contiguous<Item = Operation<F, U>>,

237:     S: Strategy,

240:     /// Set commit metadata included in the [`merkleize`](Self::merkleize) call, replacing any

243:         self.metadata = Some(metadata);

247:     /// Expand this staged batch with more reads.
248:     ///
249:     /// Existing read indices remain stable. Newly read keys are appended to the staged read set and
250:     /// assigned the returned range. Expansion does not deduplicate against previously staged keys
251:     /// and does not observe values computed for earlier staged slots but not yet passed to
252:     /// `merkleize`.
253:     pub async fn expand(
254:         self,
255:         keys: &[&U::Key],
256:     ) -> Result<(Range<usize>, Vec<Option<U::Value>>, Self), Error<F>> {
257:         let Self {
258:             staged,
259:             db,
260:             metadata,
261:         } = self;
262:         let (range, values, staged) = {
263:             let guard = db.read().await;
264:             staged.expand(keys, &guard).await?
265:         };
266:         Ok((
267:             range,
268:             values,
269:             Self {
270:                 staged,
271:                 db,
272:                 metadata,
273:             },
274:         ))
275:     }
276: }

310:         let Self {
311:             staged,
312:             db,
313:             metadata,
314:         } = self;
315:         let inner = {
316:             let guard = db.read().await;
317:             staged.merkleize(updates, upserts, metadata, &guard).await?
318:         };
319:         Ok(CurrentMerkleized { inner, db })
320:     }
321: }

1381:     /// The glue staged wrapper (`CurrentUnmerkleized::stage` -> `CurrentStaged::expand` ->
1382:     /// `CurrentStaged::merkleize`) must return the same values and root as an explicit `get_many` +
1383:     /// `write` + `merkleize`, including a staged delete, an upsert, and metadata flow (both set
1384:     /// on the staged handle via `with_metadata` and carried from before staging). This guards
1385:     /// metadata flow and db-handle pairing through the wrapper.
1386:     #[test]
1387:     fn ordered_fixed_staged_merkleize_matches_explicit_writes() {
1388:         deterministic::Runner::default().start(|context| async move {
1389:             let config = fixed_config("ordered-fixed-glue-staged", &context);
1390:             let db = <OrderedFixedDb as ManagedDb<_>>::init(context.child("db"), config)
1391:                 .await
1392:                 .unwrap();
1393:             let db = Shared::new("test", db);
1394: 
1395:             let key = |i: u64| Sha256::hash(&[&i.to_be_bytes()]);
1396:             let val = |i: u64| Sha256::hash(&[&(i + 10_000).to_be_bytes()]);
1397:             let metadata = Sha256::hash(&[b"metadata"]);
1398: 
1399:             // Seed keys 0..50 and persist them.
1400:             let mut seed = db.new_batch_for_test::<_>().await;
1401:             for i in 0..50u64 {
1402:                 seed = seed.write(key(i), Some(val(i)));
1403:             }
1404:             let merkleized = crate::stateful::db::Unmerkleized::merkleize(seed)
1405:                 .await
1406:                 .unwrap();
1407:             {
1408:                 let (slot, database) = db.write().await;
1409:                 slot.put(apply_and_finalize::<OrderedFixedDb>(database, merkleized).await);
1410:             }
1411: 
1412:             // Read set: key(1) updated, key(2) deleted, key(999) missing -> created.
1413:             let read_keys = [key(1), key(2), key(999)];
1414:             let keys: Vec<&Digest> = read_keys.iter().collect();
1415:             let indexed_updates = vec![(0, Some(val(1_000))), (1, None), (2, Some(val(1_001)))];
1416:             let upserts = vec![(key(3), Some(val(1_002)))];
1417: 
1418:             // Explicit path.
1419:             let mut explicit = db.new_batch_for_test::<_>().await;
1420:             let explicit_values = explicit.get_many(&keys).await.unwrap();
1421:             for (slot, value) in &indexed_updates {
1422:                 explicit = explicit.write(read_keys[*slot], *value);
1423:             }
1424:             for (k, v) in &upserts {
1425:                 explicit = explicit.write(*k, *v);
1426:             }
1427:             let explicit_root =
1428:                 crate::stateful::db::Unmerkleized::merkleize(explicit.with_metadata(metadata))
1429:                     .await
1430:                     .unwrap()
1431:                     .root();
1432: 
1433:             // Staged path, with metadata set on the staged handle.
1434:             let staged_batch = db.new_batch_for_test::<_>().await;
1435:             let split = 2;
1436:             let (mut staged_values, staged) = staged_batch.stage(&keys[..split]).await.unwrap();
1437:             let (range, suffix_values, staged) = staged.expand(&keys[split..]).await.unwrap();
1438:             assert_eq!(range, split..keys.len());
1439:             staged_values.extend(suffix_values);
1440:             let staged_root = staged
1441:                 .with_metadata(metadata)
1442:                 .merkleize(indexed_updates.clone(), upserts.clone())
1443:                 .await
1444:                 .unwrap()
1445:                 .root();
1446: 
1447:             assert_eq!(explicit_values, staged_values);
1448:             assert_eq!(explicit_root, staged_root);
1449: 
1450:             // Metadata set before staging must be carried through to staged merkleize.
1451:             let carried_batch = db.new_batch_for_test::<_>().await.with_metadata(metadata);
1452:             let (carried_values, staged) = carried_batch.stage(&keys).await.unwrap();
1453:             let carried_root = staged
1454:                 .merkleize(indexed_updates.clone(), upserts.clone())
1455:                 .await
1456:                 .unwrap()
1457:                 .root();
1458:             assert_eq!(explicit_values, carried_values);
1459:             assert_eq!(explicit_root, carried_root);
1460:         });
1461:     }
