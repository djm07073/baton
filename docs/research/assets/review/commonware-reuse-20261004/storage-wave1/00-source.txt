177: /// Merge two key-sorted diffs with disjoint keys into one sorted diff.
178: fn merge_sorted_diffs<K: Ord, F: Family, V>(
179:     a: DiffVec<K, F, V>,
180:     b: DiffVec<K, F, V>,
181: ) -> DiffVec<K, F, V> {
182:     let mut merged = Vec::with_capacity(a.len() + b.len());
183:     let mut a = a.into_iter().peekable();
184:     let mut b = b.into_iter().peekable();
185:     while let (Some(x), Some(y)) = (a.peek(), b.peek()) {
186:         if x.0 < y.0 {
187:             merged.push(a.next().expect("peeked"));
188:         } else {
189:             merged.push(b.next().expect("peeked"));
190:         }
191:     }
192:     merged.extend(a);
193:     merged.extend(b);
194:     merged
195: }

303: /// A speculative batch of operations whose root digest has been computed,
304: /// in contrast to [`UnmerkleizedBatch`].
305: ///
306: /// # Forking
307: ///
308: /// Multiple children can share the same parent, forming a tree:
309: ///
310: /// ```text
311: /// DB <-- B1 <-- B2 <-- B4
312: ///                \
313: ///                 B3
314: /// ```
315: ///
316: /// # Committing batches
317: ///
318: /// [`Db::apply_batch`] applies the batch and any uncommitted ancestors automatically.
319: ///
320: /// ```text
321: /// db.apply_batch(b1).await.unwrap();
322: /// db.apply_batch(b3).await.unwrap();  // Also applies b2's changes.
323: /// ```
324: ///
325: /// # Branch validity
326: ///
327: /// A `MerkleizedBatch` is a branch-scoped view rooted at a specific committed prefix of the DB,
328: /// not an immutable snapshot. Reads through the chain, constructing child batches, and applying
329: /// the batch later are only valid while every batch applied to the DB since this batch was
330: /// merkleized is an ancestor of this batch. Applying a batch from a different fork is rejected
331: /// with [`crate::qmdb::Error::StaleBatch`] (see [`crate::qmdb::batch_chain`] for more details).
332: #[allow(clippy::type_complexity)]
333: #[derive(Clone)]

398: /// Outcome of classifying one floor-raise candidate against the batch diff, ancestor
399: /// diffs, and committed snapshot.
400: ///
401: /// Classification is a pure function of the pre-raise state: at most one candidate per key
402: /// can be active (the bitmap holds exactly one set bit per committed key, and each diff or
403: /// ancestor entry resolves a key to a single location), and a move only rewrites the moved
404: /// key's own diff entry to a location above the scan tip. Classifying all candidates
405: /// against a single snapshot of the diff therefore yields the same outcomes as the
406: /// interleaved sequential walk, which lets the per-candidate work run sharded across the
407: /// strategy pool.
408: enum FloorOutcome<F: Family> {
409:     /// Not the active op for its key (or not a keyed op); leave in place.
410:     Inactive,
411:     /// Active with an existing diff entry at this index; move and rewrite it in place.
412:     MoveExisting {
413:         idx: usize,
414:         base_old_loc: Option<Location<F>>,
415:     },
416:     /// Active with no diff entry; move and stage a new entry.
417:     MoveNew { base_old_loc: Option<Location<F>> },
418: }

420: /// Streaming equivalent of [`resolve_in_ancestors`] for an ascending sequence of queries:
421: /// one cursor per key-sorted diff advances in a linear merge instead of binary-searching
422: /// each diff per key. Diffs must be ordered closest-first (the first hit wins).

424:     diffs: Vec<(&'a DiffSlice<K, F, V>, usize)>,

427: impl<'a, K: Ord, F: Family, V> DiffCursors<'a, K, F, V> {
428:     pub(crate) fn new(diffs: impl IntoIterator<Item = &'a DiffSlice<K, F, V>>) -> Self {
429:         Self {
430:             diffs: diffs.into_iter().map(|diff| (diff, 0)).collect(),
431:         }
432:     }
433: 
434:     /// Resolve `key` against the diffs (closest-first). Queries must be non-decreasing:
435:     /// cursors only advance, so an out-of-order query could miss entries.
436:     ///
437:     /// # Panics
438:     ///
439:     /// Panics on any out-of-order query that would return a wrong result (the cursor has
440:     /// already advanced past an entry at or above the query).
441:     pub(crate) fn resolve(&mut self, key: &K) -> Option<&'a DiffEntry<F, V>> {
442:         for (diff, cursor) in &mut self.diffs {
443:             assert!(
444:                 *cursor == 0 || diff[*cursor - 1].0 < *key,
445:                 "queries must be non-decreasing"
446:             );
447:             while *cursor < diff.len() && diff[*cursor].0 < *key {
448:                 *cursor += 1;
449:             }
450:             if let Some((k, entry)) = diff.get(*cursor)
451:                 && k == key
452:             {
453:                 return Some(entry);
454:             }
455:         }
456:         None
457:     }
458: }

460: /// Resolve unresolved input slots against ancestor diffs, preserving final results by original
461: /// input slot. The caller keeps `pending` in input order so DB fallthrough can do the same.
462: ///
463: /// `on_hit` is invoked (serially, in `pending` order) with each resolving diff entry, so
464: /// staged reads can record ancestor resolutions alongside the values.
465: fn resolve_pending_from_diffs<'a, K, F: Family, V: Clone + Send + Sync + 'a, S: Strategy>(
466:     pending: &[PendingRead<'a, K>],
467:     diffs: &[&'a DiffSlice<K, F, V>],
468:     strategy: &S,
469:     resolved: &mut [bool],
470:     results: &mut [Option<V>],
471:     mut on_hit: impl FnMut(usize, &DiffEntry<F, V>),
472: ) where
473:     K: Ord + Sync,
474: {
475:     if pending.is_empty() || diffs.is_empty() {
476:         return;
477:     }
478: 
479:     let resolve = |chunk: &[PendingRead<'a, K>]| -> Vec<(usize, &'a DiffEntry<F, V>)> {
480:         chunk
481:             .iter()
482:             .filter_map(|(slot, key)| {
483:                 diffs
484:                     .iter()
485:                     .find_map(|diff| lookup_sorted(diff, key))
486:                     .map(|entry| (*slot, entry))
487:             })
488:             .collect()
489:     };
490:     let hits: Vec<(usize, &'a DiffEntry<F, V>)> = strategy.run(
491:         pending.len(),
492:         || resolve(pending),
493:         || {
494:             let manual = strategy.manual();
495:             let chunk_len = pending.len().div_ceil(manual.parallelism());
496:             let chunks: Vec<_> = pending.chunks(chunk_len).collect();
497:             manual
498:                 .map_collect_vec(chunks, &resolve)
499:                 .into_iter()
500:                 .flatten()
501:                 .collect()
502:         },
503:     );
504: 
505:     for (slot, entry) in hits {
506:         resolved[slot] = true;
507:         results[slot] = entry.value().cloned();
508:         on_hit(slot, entry);
509:     }
510: }

512: /// Resolve `keys` against a local source (`local` returns `Some` when it owns the key, with the
513: /// inner `Option` distinguishing a live value from a delete) and then against `diffs`, returning
514: /// per-slot results and the slots that still need committed DB reads.
515: ///
516: /// `on_diff_hit` is invoked with each slot resolved by a diff entry (see
517: /// [`resolve_pending_from_diffs`]). Slots resolved by `local` do not report.
518: fn resolve_reads<'a, K, F: Family, V, S: Strategy>(
519:     keys: &[&'a K],
520:     local: impl Fn(&K) -> Option<Option<V>>,
521:     diffs: &[&DiffSlice<K, F, V>],
522:     strategy: &S,
523:     on_diff_hit: impl FnMut(usize, &DiffEntry<F, V>),
524: ) -> UncommittedReadResolution<'a, K, V>
525: where
526:     K: Ord + Sync,
527:     V: Clone + Send + Sync,
528: {
529:     let mut results = vec![None; keys.len()];
530:     let mut resolved = vec![false; keys.len()];
531:     let mut pending = Vec::new();
532: 
533:     for (i, key) in keys.iter().enumerate() {
534:         if let Some(value) = local(key) {
535:             results[i] = value;
536:             resolved[i] = true;
537:         } else {
538:             pending.push((i, *key));
539:         }
540:     }
541:     resolve_pending_from_diffs(
542:         &pending,
543:         diffs,
544:         strategy,
545:         &mut resolved,
546:         &mut results,
547:         on_diff_hit,
548:     );
549: 
550:     let unresolved = pending.into_iter().filter(|(i, _)| !resolved[*i]).collect();
551:     (results, unresolved)
552: }

554: /// Apply a single diff entry to the snapshot index and activity bitmap in lockstep:
555: /// install the winning `Active` location and clear the prior committed location.
556: fn apply_diff<F: Family, V, I: UnorderedIndex<Value = Location<F>>, const N: usize>(
557:     snapshot: &mut I,
558:     bitmap: &mut bitmap::Prunable<N>,
559:     key: &impl Key,
560:     entry: &DiffEntry<F, V>,
561:     base_old_loc: Option<Location<F>>,
562: ) {
563:     match entry {
564:         DiffEntry::Active { loc, .. } => match base_old_loc {
565:             Some(old) => update_known_loc::<F, _>(snapshot, key, old, *loc),
566:             None => snapshot.insert(key, *loc),
567:         },
568:         DiffEntry::Deleted { .. } => {
569:             if let Some(old) = base_old_loc {
570:                 delete_known_loc::<F, _>(snapshot, key, old);
571:             }
572:         }
573:     }
574:     if let Some(loc) = entry.loc() {
575:         bitmap.set_bit(*loc, true);
576:     }
577:     if let Some(loc) = base_old_loc {
578:         bitmap.set_bit(*loc, false);
579:     }
580: }

582: /// k-way sorted merge over diff slices in priority order. On equal keys, the lowest-indexed
583: /// stream wins and all tied cursors are advanced. Each input slice must be sorted by key.
584: struct DiffMerge<'a, K, F: Family, V> {
585:     cursors: Vec<(&'a DiffSlice<K, F, V>, usize)>,
586: }
587: 
588: impl<'a, K: Ord, F: Family, V> DiffMerge<'a, K, F, V> {
589:     fn new(streams: impl IntoIterator<Item = &'a DiffSlice<K, F, V>>) -> Self {
590:         Self {
591:             cursors: streams.into_iter().map(|s| (s, 0)).collect(),
592:         }
593:     }
594: 
595:     fn peek_key(cursor: &(&'a DiffSlice<K, F, V>, usize)) -> Option<&'a K> {
596:         cursor.0.get(cursor.1).map(|(k, _)| k)
597:     }
598: 
599:     fn next_general(&mut self) -> Option<(&'a K, &'a DiffEntry<F, V>)> {
600:         let n = self.cursors.len();
601:         let mut winner: Option<usize> = None;
602:         for level in 0..n {
603:             let Some(k) = Self::peek_key(&self.cursors[level]) else {
604:                 continue;
605:             };
606:             let better = match winner {
607:                 None => true,
608:                 Some(w) => *k < *Self::peek_key(&self.cursors[w]).unwrap(),
609:             };
610:             if better {
611:                 winner = Some(level);
612:             }
613:         }
614:         let level = winner?;
615:         let (slice, pos) = self.cursors[level];
616:         let winning_key = &slice[pos].0;
617:         for cursor in &mut self.cursors {
618:             if Self::peek_key(cursor).is_some_and(|k| k == winning_key) {
619:                 cursor.1 += 1;
620:             }
621:         }
622:         Some((&slice[pos].0, &slice[pos].1))
623:     }
624: }
625: 
626: impl<'a, K: Ord, F: Family, V> Iterator for DiffMerge<'a, K, F, V> {
627:     type Item = (&'a K, &'a DiffEntry<F, V>);
628: 
629:     fn next(&mut self) -> Option<Self::Item> {
630:         match self.cursors.len() {
631:             0 => None,
632:             1 => {
633:                 let (slice, pos) = &mut self.cursors[0];
634:                 let (k, entry) = slice.get(*pos)?;
635:                 *pos += 1;
636:                 Some((k, entry))
637:             }
638:             2 => {
639:                 let ka = Self::peek_key(&self.cursors[0]);
640:                 let kb = Self::peek_key(&self.cursors[1]);
641:                 let winner = match (ka, kb) {
642:                     (Some(a), Some(b)) => match a.cmp(b) {
643:                         Ordering::Less => 0,
644:                         Ordering::Greater => 1,
645:                         Ordering::Equal => {
646:                             self.cursors[1].1 += 1;
647:                             0
648:                         }
649:                     },
650:                     (Some(_), None) => 0,
651:                     (None, Some(_)) => 1,
652:                     (None, None) => return None,
653:                 };
654:                 let (slice, pos) = &mut self.cursors[winner];
655:                 let (k, entry) = &slice[*pos];
656:                 *pos += 1;
657:                 Some((k, entry))
658:             }
659:             _ => self.next_general(),
660:         }
661:     }
662: }

664: /// Fill `out` with up to `limit` floor-raise candidates in `[floor, tip)` under a single bitmap
665: /// read guard, returning the next `floor`.
666: fn fill_candidates<F: Family, const N: usize>(
667:     bitmap: &Shared<N>,
668:     floor: Location<F>,
669:     tip: u64,
670:     limit: usize,
671:     out: &mut Vec<Location<F>>,
672: ) -> Location<F> {
673:     Location::new(bitmap.fill_candidates(*floor, tip, limit, out))

676: /// Resolve `loc` to an op within the in-memory ancestor region
677: /// `[db_size, ancestors[0].journal_batch.size())`, walked parent-first.
678: ///
679: /// # Panics
680: ///
681: /// Panics if `loc` cannot be located in the chain: either it falls outside the region (including
682: /// when `ancestors` is empty), or the ancestor spans are non-contiguous (a bookkeeping invariant
683: /// violation).
684: fn read_op_from_ancestors<F: Family, D: Digest, U: update::Update, S: Strategy>(
685:     ancestors: &[Arc<MerkleizedBatch<F, D, U, S>>],
686:     loc: u64,
687:     db_size: u64,
688: ) -> &Operation<F, U> {
689:     // ancestors is ordered parent-first: [parent, grandparent, ...].
690:     // Each batch's items span [next_batch.size(), this_batch.size()).
691:     // The last ancestor's base is db_size (committed DB boundary).
692:     for (i, batch) in ancestors.iter().enumerate() {
693:         let batch_base = ancestors
694:             .get(i + 1)
695:             .map_or(db_size, |b| b.journal_batch.size());
696:         let batch_end = batch.journal_batch.size();
697:         if loc >= batch_base && loc < batch_end {
698:             return &batch.journal_batch.items()[(loc - batch_base) as usize];
699:         }
700:     }
701:     unreachable!("location {loc} not found in ancestor chain (db_size={db_size})")
702: }

704: /// Read helpers on [`Merkleizer`].
705: ///
706: /// # Operation-location model
707: ///
708: /// The operation space is divided into three contiguous regions:
709: ///
710: /// ```text
711: ///  [0 ........... db_size)  [db_size ..... base_size)  [base_size .. base_size+len)
712: ///   committed (on disk)     ancestors (in mem)          this batch (in mem)
713: /// ```
714: ///
715: /// `db_size` is the boundary between disk and in-memory ancestors. It equals the original DB size
716: /// when the full ancestor chain is alive, or a higher value if ancestors were freed (see
717: /// `into_parts`). For batches created directly from the DB (no uncommitted ancestors), the ancestor
718: /// region is empty (`db_size == base_size`).
719: ///
720: /// # Contract for all read methods
721: ///
722: /// Callers must pass a `loc` that is a valid operation location: specifically `loc < base_size +
723: /// batch_ops.len()` (i.e., within one of the three regions). Passing an out-of-range `loc` may
724: /// panic (via `batch_ops` indexing or the ancestor-chain walk) or result in a disk-read error.
725: /// In-memory locations are resolved synchronously; only disk locations await the `reader`.
726: impl<F: Family, H, U, S: Strategy> Merkleizer<F, H, U, S>
727: where
728:     U: update::Update,
729:     H: Hasher,
730:     Operation<F, U>: Codec,
731: {
732:     /// Returns `Some(op)` if `loc` falls in the batch or ancestor regions, and `None` when `loc` is
733:     /// in the committed region (`loc < db_size`).
734:     fn try_read_op_from_uncommitted(
735:         &self,
736:         loc: Location<F>,
737:         batch_ops: &[Operation<F, U>],
738:     ) -> Option<Operation<F, U>> {
739:         let loc = *loc;
740: 
741:         if loc >= self.base_state.size {
742:             return Some(batch_ops[(loc - *self.base_state.size) as usize].clone());
743:         }
744: 
745:         if loc >= self.db_state.size {
746:             return Some(read_op_from_ancestors(&self.ancestors, loc, *self.db_state.size).clone());
747:         }
748: 
749:         None
750:     }
751: 
752:     /// Whether `locations` is strictly ascending and entirely within the committed region,
753:     /// the shape a single batched reader call serves with no in-memory resolution.
754:     fn all_committed_ascending(&self, locations: &[Location<F>]) -> bool {
755:         locations.is_sorted_by(|a, b| a < b)
756:             && locations
757:                 .last()
758:                 .is_some_and(|last| **last < self.db_state.size)
759:     }
760: 
761:     /// Read multiple operations by location, preserving the caller's order and permitting
762:     /// duplicates.
763:     ///
764:     /// Batch and ancestor regions resolve in memory. All committed locations are served by
765:     /// one batched read, which serves page-cache hits under a single lock acquisition per
766:     /// section instead of paying a cache lock acquisition per location.
767:     async fn read_ops<R: Contiguous<Item = Operation<F, U>>>(
768:         &self,
769:         locations: &[Location<F>],
770:         batch_ops: &[Operation<F, U>],
771:         reader: &R,
772:     ) -> Result<Vec<Operation<F, U>>, crate::qmdb::Error<F>> {
773:         // Fast path: a strictly ascending batch entirely within the committed region needs no
774:         // in-memory resolution, reordering, or per-location bookkeeping, so the positions can
775:         // be handed to the reader directly. Depth-0 mutation reads take this path. Floor-raise
776:         // candidate reads hit the same predicate in read_ops_sharded and reach here only when
777:         // candidates cross into the uncommitted region.
778:         if self.all_committed_ascending(locations) {
779:             let positions: Vec<u64> = locations.iter().map(|loc| **loc).collect();
780:             return Ok(reader.read_many(&positions).await?);
781:         }
782: 
783:         // Resolve the in-memory regions synchronously.
784:         let mut results: Vec<Option<Operation<F, U>>> = locations
785:             .iter()
786:             .map(|loc| self.try_read_op_from_uncommitted(*loc, batch_ops))
787:             .collect();
788: 
789:         // Batch-read committed locations. Reader::read_many requires sorted, unique positions.
790:         let committed: Vec<(usize, u64)> = locations
791:             .iter()
792:             .zip(results.iter())
793:             .enumerate()
794:             .filter_map(|(idx, (loc, resolved))| resolved.is_none().then_some((idx, **loc)))
795:             .collect();
796:         if committed.is_empty() {
797:             return Ok(results.into_iter().map(Option::unwrap).collect());
798:         }
799: 
800:         // Batches reaching here contain uncommitted locations or arrived unsorted, but the
801:         // committed subset is often still presorted (e.g. floor-raise candidates that cross
802:         // the committed boundary), so the sort is worth skipping when possible.
803:         let mut positions: Vec<u64> = committed.iter().map(|(_, loc)| *loc).collect();
804:         let presorted = positions.is_sorted_by(|a, b| a < b);
805:         if !presorted {
806:             positions.sort_unstable();
807:             positions.dedup();
808:         }
809:         let read = reader.read_many(&positions).await?;
810: 
811:         // Merge read results back in order.
812:         for (idx, loc) in committed {
813:             // `positions` is sorted and deduped, and `loc` came from it before deduping, so
814:             // binary search must find the matching read_many result.
815:             let result_idx = positions
816:                 .binary_search(&loc)
817:                 .expect("read result missing for requested location");
818:             results[idx] = Some(read[result_idx].clone());
819:         }
820:         Ok(results
821:             .into_iter()
822:             .map(|r| r.expect("operation should be resolved"))
823:             .collect())
824:     }
825: 
826:     /// Like [`read_ops`](Self::read_ops), but returns chunk-partitioned results whose
827:     /// concatenation preserves `locations` order. A strictly ascending batch entirely
828:     /// within the committed region (the typical floor-raise candidate read) stays
829:     /// partitioned as the reader probed it, skipping serial reassembly on the calling
830:     /// task. Other shapes resolve through [`read_ops`](Self::read_ops) as a single chunk.
831:     async fn read_ops_sharded<E, C>(
832:         &self,
833:         locations: &[Location<F>],
834:         batch_ops: &[Operation<F, U>],
835:         reader: &authenticated::Journal<F, E, C, H, S>,
836:     ) -> Result<Vec<Vec<Operation<F, U>>>, crate::qmdb::Error<F>>
837:     where
838:         E: Context,
839:         C: Contiguous<Item = Operation<F, U>>,
840:         Operation<F, U>: Codec,
841:     {
842:         if self.all_committed_ascending(locations) {
843:             let positions: Vec<u64> = locations.iter().map(|loc| **loc).collect();
844:             return Ok(reader.read_many_sharded(&positions).await?);
845:         }
846:         Ok(vec![self.read_ops(locations, batch_ops, reader).await?])
847:     }
848: 
849:     /// Gather existing-key locations for all keys in `mutations`.
850:     ///
851:     /// For each mutation key, checks the ancestor diffs first (returning the uncommitted
852:     /// location for Active entries, skipping Deleted entries). Keys not in the ancestor diffs
853:     /// fall back to the committed DB snapshot.
854:     ///
855:     /// When `include_active_collision_siblings` is true, Active entries also scan the snapshot
856:     /// bucket for collision siblings (other keys sharing the same translated-key bucket). The
857:     /// ordered path needs these so their `next_key` pointers are rewritten when a sibling is
858:     /// deleted; the unordered path can skip them.
859:     fn gather_existing_locations<E, C, I, const N: usize>(
860:         &self,
861:         mutations: &BTreeMap<U::Key, Option<U::Value>>,
862:         db: &Db<F, E, C, I, H, U, N, S>,
863:         include_active_collision_siblings: bool,
864:     ) -> Vec<Location<F>>
865:     where
866:         E: Context,
867:         C: Contiguous<Item = Operation<F, U>>,
868:         I: UnorderedIndex<Value = Location<F>>,
869:     {
870:         // Extra slack (*3/2) avoids re-allocations when index collisions cause more than one
871:         // location per key.
872:         let mut locations = Vec::with_capacity(mutations.len() * 3 / 2);
873:         if self.ancestors.is_empty() {
874:             for key in mutations.keys() {
875:                 locations.extend(db.snapshot.get(key).copied());
876:             }
877:         } else {
878:             let mut ancestors = DiffCursors::new(self.ancestors.iter().map(|a| a.diff.as_slice()));
879:             for key in mutations.keys() {
880:                 match ancestors.resolve(key) {
881:                     Some(DiffEntry::Deleted { .. }) => {
882:                         // Stale; handled via extract_parent_deleted_creates.
883:                     }
884:                     Some(DiffEntry::Active {
885:                         loc, base_old_loc, ..
886:                     }) => {
887:                         locations.push(*loc);
888:                         if include_active_collision_siblings {
889:                             locations.extend(
890:                                 db.snapshot
891:                                     .get(key)
892:                                     .copied()
893:                                     .filter(move |loc| Some(*loc) != *base_old_loc),
894:                             );
895:                         }
896:                     }
897:                     None => {
898:                         locations.extend(db.snapshot.get(key).copied());
899:                     }
900:                 }
901:             }
902:         }
903:         db.strategy().sort_by(&mut locations, |a, b| a.cmp(b));
904:         locations.dedup();
905:         locations
906:     }
907: 
908:     /// Extract keys that were deleted by a parent batch but are being
909:     /// re-created by this child batch. Removes those keys from `mutations`
910:     /// and returns `(key, value, base_old_loc)` entries.
911:     #[allow(clippy::type_complexity)]
912:     fn extract_parent_deleted_creates(
913:         &self,
914:         mutations: &mut BTreeMap<U::Key, Option<U::Value>>,
915:     ) -> Vec<(U::Key, U::Value, Option<Location<F>>)> {
916:         if self.ancestors.is_empty() {
917:             return Vec::new();
918:         }
919:         let mut ancestors = DiffCursors::new(self.ancestors.iter().map(|a| a.diff.as_slice()));
920:         let mut creates = Vec::new();
921:         mutations.retain(|key, value| {
922:             if let Some(DiffEntry::Deleted { base_old_loc }) = ancestors.resolve(key)
923:                 && let Some(v) = value.take()
924:             {
925:                 creates.push((key.clone(), v, *base_old_loc));
926:                 return false;
927:             }
928:             true
929:         });
930:         creates
931:     }

933:     /// Shared final phases of merkleization: floor raise, CommitFloor, journal
934:     /// merkleize, diff merge, and `MerkleizedBatch` construction.
935:     ///
936:     /// `diff` may arrive in any order: it is key-sorted on the strategy pool, overlapping the
937:     /// first floor-raise candidate read. `superseded_locs` holds the committed locations
938:     /// superseded by `diff` (every `Some` `base_old_loc`), in any order. The floor raise
939:     /// skips re-reading them. `prefetched` optionally holds committed-prefix candidates the
940:     /// caller gathered and read ahead of time, consumed by the raise before scanning live.
941:     #[allow(clippy::too_many_arguments)]
942:     async fn finish<E, C, I, const N: usize>(
943:         self,
944:         mut ops: Vec<Operation<F, U>>,
945:         mut diff: DiffVec<U::Key, F, U::Value>,
946:         mut superseded_locs: Vec<Location<F>>,
947:         active_keys_delta: isize,
948:         user_steps: u64,
949:         metadata: Option<U::Value>,
950:         mut prefetched: Option<PrefetchedCandidates<F, U>>,
951:         mut fill_candidates: impl FnMut(Location<F>, u64, usize, &mut Vec<Location<F>>) -> Location<F>,
952:         db: &Db<F, E, C, I, H, U, N, S>,
953:     ) -> Result<Arc<MerkleizedBatch<F, H::Digest, U, S>>, crate::qmdb::Error<F>>
954:     where
955:         E: Context,
956:         C: Contiguous<Item = Operation<F, U>>,
957:         I: UnorderedIndex<Value = Location<F>>,
958:     {
959:         // Floor raise.
960:         // Steps = user_steps + 1 (+1 for previous commit becoming inactive).
961:         let total_steps = user_steps + 1;
962:         let total_active_keys = self.base_active_keys as isize + active_keys_delta;
963:         let mut floor = self.base_inactivity_floor_loc;
964: 
965:         // Key-sort the diff as one job on the strategy: candidate classification (after the
966:         // first floor-raise read below) is the earliest consumer that needs it sorted, so the
967:         // sort overlaps the candidate gathering and read instead of the calling task. An
968:         // empty diff is already sorted and skips the job. While the job runs, `diff` is
969:         // empty. It is replaced by the sorted diff at the first `diff_sort` await.
970:         let mut diff_sort = None;
971:         if !diff.is_empty() {
972:             let unsorted = mem::take(&mut diff);
973:             diff_sort = Some(db.strategy().spawn(unsorted.len(), move |strategy| {
974:                 let mut diff = unsorted;
975:                 strategy.sort_by(&mut diff, |a, b| a.0.cmp(&b.0));
976:                 diff
977:             }));
978:         }
979: 
980:         // New diff entries for keys moved by the floor raise, merged into `diff` below.
981:         let mut floor_diff = Vec::new();
982:         if total_active_keys > 0 {
983:             // Floor raise: advance the inactivity floor by `total_steps` active operations.
984:             // `fixed_tip` prevents scanning into floor-raise moves just appended.
985:             let strategy = db.strategy();
986:             let fixed_tip = *self.base_state.size + ops.len() as u64;
987:             let mut moved = 0u64;
988:             let mut scan_from = floor;
989:             floor_diff.reserve(total_steps as usize);
990: 
991:             // Locations are unique (each committed location belongs to exactly one key), so a
992:             // presorted collection needs neither the sort nor the dedup.
993:             if !superseded_locs.is_sorted_by(|a, b| a < b) {
994:                 strategy.sort_by(&mut superseded_locs, |a, b| a.cmp(b));
995:                 superseded_locs.dedup();
996:             }
997: 
998:             // The raise appends at most `total_steps` moved ops plus the CommitFloor. Reserve
999:             // once instead of growing mid-loop.
1000:             ops.reserve(total_steps as usize + 1);
1001: 
1002:             // `fill_candidates` yields ascending locations, so superseded checks advance a
1003:             // monotonic cursor.
1004:             let mut superseded_cursor = 0;
1005: 
1006:             // Scan active operations in `[floor, fixed_tip)` and move them to the tip.
1007:             while moved < total_steps {
1008:                 // Collect candidates, capped by the number of active ops still needed.
1009:                 // `scan_from` tracks prefetch progress separately from `floor`, so
1010:                 // early exit cannot leave `floor` past unprocessed candidates.
1011:                 let limit = (total_steps - moved) as usize;
1012: 
1013:                 // Consume the prefetched committed prefix whole: it was gathered from the
1014:                 // same floor with the same bitmap, so it is a prefix of the sequence the
1015:                 // live scan would produce, and `next_scan` hands the live scan its
1016:                 // continuation point. Handing the raise more candidates than `limit` is
1017:                 // outcome-identical to fetching them across rounds: classification is pure
1018:                 // per candidate and the apply loop stops advancing once enough ops moved.
1019:                 let (mut candidates, pf_shards) = match prefetched.take() {
1020:                     Some(pf) => {
1021:                         scan_from = pf.next_scan;
1022:                         (pf.locs, pf.shards)
1023:                     }
1024:                     None => (Vec::with_capacity(limit), Vec::new()),
1025:                 };
1026:                 if candidates.len() < limit {
1027:                     scan_from = fill_candidates(
1028:                         scan_from,
1029:                         fixed_tip,
1030:                         limit - candidates.len(),
1031:                         &mut candidates,
1032:                     );
1033:                 }
1034:                 if candidates.is_empty() {
1035:                     break;
1036:                 }
1037: 
1038:                 // The `sorted_contains` cursor relies on the candidate sequence ascending
1039:                 // across the whole raise. `floor` is one past the last processed candidate.
1040:                 assert!(candidates[0] >= floor);
1041:                 assert!(candidates.is_sorted_by(|a, b| a < b));
1042: 
1043:                 // `read_candidates` omits locations already superseded by this diff, saving
1044:                 // their read. Keep `resolved` and `outcomes` in that filtered order, then
1045:                 // walk `candidates` below so superseded locations still advance the floor in
1046:                 // scan order. Prefetched candidates skip the filter -- their ops were read
1047:                 // ahead of time, and a superseded candidate's key always resolves in the
1048:                 // diff to a different location, classifying it `Inactive`.
1049:                 let pf_count: usize = pf_shards.iter().map(Vec::len).sum();
1050:                 assert!(pf_count <= candidates.len());
1051:                 let mut read_candidates: Vec<Location<F>> = Vec::with_capacity(candidates.len());
1052:                 read_candidates.extend_from_slice(&candidates[..pf_count]);
1053:                 for candidate in &candidates[pf_count..] {
1054:                     if !sorted_contains(&superseded_locs, &mut superseded_cursor, candidate) {
1055:                         read_candidates.push(*candidate);
1056:                     }
1057:                 }
1058:                 let (resolved, outcomes): (_, Vec<FloorOutcome<F>>) = if read_candidates.is_empty()
1059:                 {
1060:                     (Vec::new(), Vec::new())
1061:                 } else {
1062:                     // Batch-read candidates: page-cache hits are served by one batched read,
1063:                     // disk misses are fetched concurrently. Prefetched shards enter as the
1064:                     // reader probed them, ahead of the live suffix's read.
1065:                     let live = &read_candidates[pf_count..];
1066:                     let mut resolved = pf_shards;
1067:                     if !live.is_empty() {
1068:                         resolved.extend(self.read_ops_sharded(live, &ops, &db.log).await?);
1069:                     }
1070: 
1071:                     // Classification is the first consumer of the sorted diff. By now the
1072:                     // sort has overlapped the fill and read above.
1073:                     if let Some(job) = diff_sort.take() {
1074:                         diff = job.await;
1075:                     }
1076: 
1077:                     // Classify read candidates against the pre-raise state (see
1078:                     // [`FloorOutcome`]). Revalidation is required even for candidates whose
1079:                     // committed bitmap bit is set: an uncommitted ancestor diff may supersede
1080:                     // the committed location, and that is not reflected in the bitmap.
1081:                     let classify = |candidate: Location<F>, op: &Operation<F, U>| {
1082:                         let Some(key) = op.key() else {
1083:                             return FloorOutcome::Inactive; // CommitFloor and other non-keyed ops
1084:                         };
1085:                         match diff.binary_search_by(|(k, _)| k.cmp(key)) {
1086:                             Ok(idx) => {
1087:                                 let entry = &diff[idx].1;
1088:                                 if entry.loc() == Some(candidate) {
1089:                                     FloorOutcome::MoveExisting {
1090:                                         idx,
1091:                                         base_old_loc: entry.base_old_loc(),
1092:                                     }
1093:                                 } else {
1094:                                     FloorOutcome::Inactive
1095:                                 }
1096:                             }
1097:                             Err(_) => resolve_in_ancestors(&self.ancestors, key).map_or_else(
1098:                                 || {
1099:                                     if db.snapshot.get(key).any(|&l| l == candidate) {
1100:                                         FloorOutcome::MoveNew {
1101:                                             base_old_loc: Some(candidate),
1102:                                         }
1103:                                     } else {
1104:                                         FloorOutcome::Inactive
1105:                                     }
1106:                                 },
1107:                                 |entry| {
1108:                                     if entry.loc() == Some(candidate) {
1109:                                         FloorOutcome::MoveNew {
1110:                                             base_old_loc: entry.base_old_loc(),
1111:                                         }
1112:                                     } else {
1113:                                         FloorOutcome::Inactive
1114:                                     }
1115:                                 },
1116:                             ),
1117:                         }
1118:                     };
1119: 
1120:                     // Classify each candidate against the pre-raise state, in candidate order.
1121:                     let outcomes: Vec<FloorOutcome<F>> = strategy.map_collect_vec(
1122:                         zip_eq(read_candidates.iter(), resolved.iter().flatten()),
1123:                         |(loc, op)| classify(*loc, op),
1124:                     );
1125:                     (resolved, outcomes)
1126:                 };
1127: 
1128:                 // Apply in candidate order, moving active ops to the tip. `read_candidates`
1129:                 // preserves candidate order, so a candidate that does not match the next
1130:                 // pending read was superseded and only advances the floor.
1131:                 let mut outcomes = outcomes.into_iter();
1132:                 let mut reads = resolved.into_iter().flatten();
1133:                 let mut pending = read_candidates.iter().peekable();
1134:                 for candidate in candidates {
1135:                     floor = candidate + 1;
1136:                     if pending.next_if(|&&pending| pending == candidate).is_none() {
1137:                         continue;
1138:                     }
1139:                     let op = reads.next().expect("one read per candidate");
1140:                     let outcome = outcomes.next().expect("one outcome per read candidate");
1141:                     match outcome {
1142:                         FloorOutcome::Inactive => continue,
1143:                         FloorOutcome::MoveExisting { idx, base_old_loc } => {
1144:                             let new_loc = self.base_state.size + ops.len() as u64;
1145:                             let value = extract_update_value(&op);
1146:                             ops.push(op);
1147:                             diff[idx].1 = DiffEntry::Active {
1148:                                 value,
1149:                                 loc: new_loc,
1150:                                 base_old_loc,
1151:                             };
1152:                         }
1153:                         FloorOutcome::MoveNew { base_old_loc } => {
1154:                             let key = op.key().cloned().expect("moved op has a key");
1155:                             let new_loc = self.base_state.size + ops.len() as u64;
1156:                             let value = extract_update_value(&op);
1157:                             ops.push(op);
1158:                             floor_diff.push((
1159:                                 key,
1160:                                 DiffEntry::Active {
1161:                                     value,
1162:                                     loc: new_loc,
1163:                                     base_old_loc,
1164:                                 },
1165:                             ));
1166:                         }
1167:                     }
1168:                     moved += 1;
1169:                     if moved >= total_steps {
1170:                         break;
1171:                     }
1172:                 }
1173:             }
1174:         } else {
1175:             // DB is empty after this batch; raise floor to tip.
1176:             floor = self.base_state.size + ops.len() as u64;
1177:             debug!(tip = ?floor, "db is empty, raising floor to tip");
1178:         }
1179: 
1180:         // The floor raise may have exited without classifying any candidate (or been skipped
1181:         // entirely). Every path below needs the sorted diff.
1182:         if let Some(job) = diff_sort.take() {
1183:             diff = job.await;
1184:         }
1185: 
1186:         // Merge the floor raise's new diff entries as one job on the strategy: nothing below
1187:         // reads `diff` until after the journal merkleization, so the merge overlaps the
1188:         // hashing instead of the calling task. `floor_diff` only accumulates keys that were
1189:         // not already present in `diff` (a key can only be moved once during this floor raise
1190:         // because, after it is moved, its new location lies above `fixed_tip` and the scan
1191:         // never revisits it), so the merge inputs are disjoint.
1192:         let mut diff_merge = None;
1193:         if !floor_diff.is_empty() {
1194:             let merge_len = floor_diff.len() + diff.len();
1195:             diff_merge = Some(db.strategy().spawn(merge_len, move |strategy| {
1196:                 let mut floor_diff = floor_diff;
1197:                 strategy.sort_by(&mut floor_diff, |a, b| a.0.cmp(&b.0));
1198:                 let diff = merge_sorted_diffs(diff, floor_diff);
1199:                 assert!(diff.is_sorted_by(|a, b| a.0 < b.0));
1200:                 diff
1201:             }));
1202:             diff = Vec::new();
1203:         }
1204: 
1205:         // CommitFloor operation.
1206:         let commit_loc = self.base_state.size + ops.len() as u64;
1207:         ops.push(Operation::CommitFloor(metadata, floor));
1208: 
1209:         // Merkleize the journal batch.
1210:         // The journal batch was created eagerly at batch construction time and its
1211:         // parent already contains all prior batches' Merkle state, so we only
1212:         // add THIS batch's operations. Parent operations are never re-cloned,
1213:         // re-encoded, or re-hashed.
1214:         let leaves = self.base_state.size + ops.len() as u64;
1215:         let inactive_peaks = db.inactive_peaks(leaves, floor);
1216: 
1217:         // Leaf and node hashing dominate merkleization, so run them as one job through the
1218:         // strategy (see `Journal::merkleize`).
1219:         let (journal, root) = db
1220:             .log
1221:             .merkleize(self.journal_batch, ops, inactive_peaks)
1222:             .await?;
1223:         if let Some(job) = diff_merge.take() {
1224:             diff = job.await;
1225:         }
1226: 
1227:         // A retained ancestor's `base_old_loc` traces back to the DB boundary at which its chain
1228:         // was originally created. If an older committed prefix has since dropped out of the Weak
1229:         // chain, resolve keys touched on both sides of that boundary to their location after the
1230:         // dropped prefix. Keeping only the intersection avoids retaining the prefix's value diffs.
1231:         let ancestor_base_locs = self.ancestors.last().map_or_else(Vec::new, |oldest| {
1232:             if oldest.ancestor_diffs.iter().all(|diff| diff.is_empty()) {
1233:                 return Vec::new();
1234:             }
1235:             let mut dropped =
1236:                 DiffCursors::new(oldest.ancestor_diffs.iter().map(|diff| diff.as_slice()));
1237:             DiffMerge::new(
1238:                 self.ancestors
1239:                     .iter()
1240:                     .map(|ancestor| ancestor.diff.as_slice()),
1241:             )
1242:             .filter_map(|(key, _)| dropped.resolve(key).map(|entry| (key.clone(), entry.loc())))
1243:             .collect()
1244:         });
1245:         let ancestor_diffs: Vec<_> = self.ancestors.iter().map(|a| Arc::clone(&a.diff)).collect();
1246:         let ancestors: Vec<_> = self
1247:             .ancestors
1248:             .iter()
1249:             .map(|a| batch_chain::AncestorBounds {
1250:                 floor: a.bounds.inactivity_floor,
1251:                 state: a.commitment(),
1252:             })
1253:             .collect();
1254: 
1255:         assert!(total_active_keys >= 0, "active_keys underflow");
1256:         Ok(Arc::new(MerkleizedBatch {
1257:             journal_batch: journal,
1258:             diff: Arc::new(diff),
1259:             parent: self.ancestors.first().map(Arc::downgrade),
1260:             total_active_keys: total_active_keys as usize,
1261:             ancestor_diffs,
1262:             ancestor_base_locs,
1263:             bounds: batch_chain::Bounds {
1264:                 base: self.base_state,
1265:                 db: self.db_state,
1266:                 tip: Commitment::new(commit_loc + 1, root),
1267:                 ancestors,
1268:                 inactivity_floor: floor,
1269:             },
1270:         }))
1271:     }
1272: }

1312: impl<F: Family, H, U, S: Strategy> Staged<F, H, U, S>
1313: where
1314:     U: update::Update,
1315:     H: Hasher,
1316:     Operation<F, U>: Codec,
1317: {
1318:     /// Expand this staged batch with more reads.
1319:     ///
1320:     /// Existing read indices remain stable. Newly read keys are appended to the staged read set and
1321:     /// assigned the returned range. The returned values are in the same order as `keys`.
1322:     ///
1323:     /// Expansion does not deduplicate against previously staged keys. Reading the same key again
1324:     /// creates another staged slot in the returned range. If both slots are later updated,
1325:     /// [`merkleize`](Staged::merkleize) applies the update list's normal last-write-wins
1326:     /// semantics.
1327:     ///
1328:     /// Expansion reads through the underlying batch, ancestor batches, and committed database state.
1329:     /// Values the caller has computed for earlier staged slots are not visible until they are passed
1330:     /// to [`merkleize`](Staged::merkleize). Callers that need speculative read-your-writes behavior
1331:     /// should maintain their own overlay while deciding which staged slots to update.
1332:     #[allow(clippy::type_complexity)]
1333:     #[tracing::instrument(
1334:         name = "qmdb.any.batch.expand",
1335:         level = "info",
1336:         skip_all,
1337:         fields(keys = keys.len() as u64, staged = self.keys.len() as u64),
1338:     )]
1339:     pub async fn expand<E, C, I, const N: usize>(
1340:         mut self,
1341:         keys: &[&U::Key],
1342:         db: &Db<F, E, C, I, H, U, N, S>,
1343:     ) -> Result<(Range<usize>, Vec<Option<U::Value>>, Self), crate::qmdb::Error<F>>
1344:     where
1345:         E: Context,
1346:         C: Contiguous<Item = Operation<F, U>>,
1347:         I: UnorderedIndex<Value = Location<F>> + 'static,
1348:     {
1349:         let start = self.keys.len();
1350:         let end = start
1351:             .checked_add(keys.len())
1352:             .expect("staged read index overflow");
1353:         let (values, keys, mut resolutions) = self.batch.stage_reads(keys, db).await?;
1354:         self.keys.extend(keys);
1355:         self.resolutions.append(&mut resolutions);
1356:         Ok((start..end, values, self))
1357:     }
1358: 
1359:     fn apply_upserts(
1360:         mut batch: UnmerkleizedBatch<F, H, U, S>,
1361:         upserts: Vec<(U::Key, Option<U::Value>)>,
1362:     ) -> UnmerkleizedBatch<F, H, U, S> {
1363:         for (key, value) in upserts {
1364:             batch = batch.write(key, value);
1365:         }
1366:         batch
1367:     }
1368: 
1369:     /// Resolve the caller's updates and upserts against the staged read set, returning the
1370:     /// underlying batch (with fallback mutations recorded) and the staged updates to consume at
1371:     /// merkleize.
1372:     ///
1373:     /// Each update is `(read_index, value)`, where `read_index` is the position of the key in the
1374:     /// staged read set: the initial [`stage`](UnmerkleizedBatch::stage) input followed by any
1375:     /// [`expand`](Staged::expand) inputs. `value` is `Some(v)` for an upsert or `None` for a
1376:     /// delete. Duplicate keys retain last-write-wins semantics according to the update order.
1377:     /// Upserts are `(key, value)` writes (`None` deletes) for keys outside the staged read set.
1378:     /// Upserts are applied last. If a caller passes an overlapping key, the upsert follows normal
1379:     /// `write` semantics and wins.
1380:     ///
1381:     /// Location-resolved updates (committed, or ancestor-diff when the update kind stages
1382:     /// those -- see [`update::Update::STAGES_ANCESTORS`]) reuse the staged location. Resolved
1383:     /// deletes reuse it only when [`update::Update::STAGES_DELETES`] is set (the unordered
1384:     /// kind). Unresolved keys (missing from committed state, resolved through this batch's
1385:     /// own mutations, or ancestor-resolved for a kind that does not stage those) always fall
1386:     /// back to normal mutations.
1387:     ///
1388:     /// # Panics
1389:     ///
1390:     /// Panics if any update's `read_index` is out of the staged read range.
1391:     pub(crate) fn resolve_updates(
1392:         self,
1393:         updates: Vec<(usize, Option<U::Value>)>,
1394:         upserts: Vec<(U::Key, Option<U::Value>)>,
1395:         strategy: &S,
1396:     ) -> (UnmerkleizedBatch<F, H, U, S>, StagedUpdates<F, U>) {
1397:         let Self {
1398:             mut batch,
1399:             keys,
1400:             mut resolutions,
1401:         } = self;
1402:         let mut staged_updates = StagedUpdates::<F, U>::new();
1403:         if updates.is_empty() {
1404:             return (Self::apply_upserts(batch, upserts), staged_updates);
1405:         }
1406: 
1407:         // Resolve last-write-wins per distinct key: a forward walk keyed on the staged key leaves
1408:         // each key's final write, the same winner as a newest-first scan. Later writes overwrite
1409:         // their key's entry in place, so `winners` holds one entry per distinct key written. A
1410:         // distinct key needs a staged slot, so the staged read set caps the reservation that a
1411:         // duplicate-heavy update list would otherwise inflate.
1412:         let capacity = updates.len().min(keys.len());
1413:         let mut winner_of: AHashMap<&U::Key, usize> = AHashMap::with_capacity(capacity);
1414:         let mut winners: Vec<Option<(usize, Option<U::Value>)>> = Vec::with_capacity(capacity);
1415:         for (slot, value) in updates {
1416:             assert!(slot < keys.len(), "update index out of staged read range");
1417:             match winner_of.entry(&keys[slot]) {
1418:                 hash_map::Entry::Vacant(vacant) => {
1419:                     vacant.insert(winners.len());
1420:                     winners.push(Some((slot, value)));
1421:                 }
1422:                 hash_map::Entry::Occupied(occupied) => {
1423:                     winners[*occupied.get()] = Some((slot, value))
1424:                 }
1425:             }
1426:         }
1427: 
1428:         // Upserts are applied last and win over overlapping staged updates. Only updated keys can
1429:         // overlap, so this probes the winner index rather than the whole staged read set.
1430:         for (key, _) in &upserts {
1431:             if let Some(&entry) = winner_of.get(key) {
1432:                 winners[entry] = None;
1433:             }
1434:         }
1435:         drop(winner_of);
1436: 
1437:         // Split the winners: updates whose slot resolved to a location become staged
1438:         // updates, the rest fall back to batch mutations. A surviving staged write must not
1439:         // also emit an older batch mutation for the same key, so it is removed here. The
1440:         // probe is skipped when the batch had no mutations before this call: each distinct
1441:         // key is visited at most once (winners are per key), so a staged winner can never
1442:         // chase a fallback inserted by this same loop.
1443:         let had_mutations = !batch.mutations.is_empty();
1444:         let mut order: Vec<(Location<F>, usize)> = Vec::with_capacity(winners.len());
1445:         for (entry, winner) in winners.iter_mut().enumerate() {
1446:             let Some((slot, value)) = winner else {
1447:                 continue;
1448:             };
1449:             let key = &keys[*slot];
1450:             match &resolutions[*slot] {
1451:                 Some((sloc, _)) if value.is_some() || U::STAGES_DELETES => {
1452:                     if had_mutations {
1453:                         batch.mutations.remove(key);
1454:                     }
1455:                     order.push((sloc.loc(), entry));
1456:                 }
1457:                 _ => {
1458:                     let (_, value) = winner.take().expect("winner checked above");
1459:                     batch.mutations.insert(key.clone(), value);
1460:                 }
1461:             }
1462:         }
1463: 
1464:         // Locations are unique after last-write-wins dedup (each key resolves to exactly one
1465:         // location, committed or ancestor), so the parallel sort is deterministic. Sorting
1466:         // compact `(location, winner)` pairs instead of the staged tuples keeps its memory
1467:         // traffic low. The tuples are then drained in sorted order, moving each winner's
1468:         // payload and value instead of cloning them.
1469:         strategy.sort_by(&mut order, |a, b| a.0.cmp(&b.0));
1470:         staged_updates = order
1471:             .iter()
1472:             .map(|&(_, entry)| {
1473:                 let (slot, value) = winners[entry]
1474:                     .take()
1475:                     .expect("winner recorded for staged slot");
1476:                 let (sloc, payload) = resolutions[slot].take().expect("resolution checked above");
1477:                 (keys[slot].clone(), sloc, payload, value)
1478:             })
1479:             .collect();
1480:         (Self::apply_upserts(batch, upserts), staged_updates)
1481:     }
1482: }

1484: impl<F: Family, K, V, H, S: Strategy> Staged<F, H, update::Unordered<K, V>, S>
1485: where
1486:     K: Key,
1487:     V: ValueEncoding,
1488:     H: Hasher,
1489:     Operation<F, update::Unordered<K, V>>: Codec,
1490: {
1491:     /// Record updates for staged reads and upserts for unread keys, then merkleize.
1492:     ///
1493:     /// Consumes the staged handle and write vectors. Call [`expand`](Staged::expand) before this
1494:     /// method if more keys must be read into the staged index space.
1495:     ///
1496:     /// A `Some` value is an upsert. `None` is a delete. Update indices refer to the staged read
1497:     /// set: the initial [`stage`](UnmerkleizedBatch::stage) input followed by any
1498:     /// [`expand`](Staged::expand) ranges. `metadata` is committed with the returned batch.
1499:     ///
1500:     /// # Panics
1501:     ///
1502:     /// Panics if any update's `read_index` is out of the staged read range.
1503:     #[allow(clippy::type_complexity)]
1504:     #[tracing::instrument(
1505:         name = "qmdb.any.unordered.batch.merkleize.staged",
1506:         level = "info",
1507:         skip_all,
1508:         fields(updates = updates.len() as u64, upserts = upserts.len() as u64),
1509:     )]
1510:     pub async fn merkleize<E, C, I, const N: usize>(
1511:         self,
1512:         updates: Vec<(usize, Option<V::Value>)>,
1513:         upserts: Vec<(K, Option<V::Value>)>,
1514:         metadata: Option<V::Value>,
1515:         db: &Db<F, E, C, I, H, update::Unordered<K, V>, N, S>,
1516:     ) -> Result<Arc<MerkleizedBatch<F, H::Digest, update::Unordered<K, V>, S>>, crate::qmdb::Error<F>>
1517:     where
1518:         E: Context,
1519:         C: Mutable<Item = Operation<F, update::Unordered<K, V>>>,
1520:         I: UnorderedIndex<Value = Location<F>>,
1521:     {
1522:         let (batch, staged_updates, prefetched) = self
1523:             .resolve_updates_prefetched(updates, upserts, db, |floor, tip, limit, out| {
1524:                 fill_candidates(&db.bitmap, floor, tip, limit, out)
1525:             })
1526:             .await?;
1527:         batch
1528:             .merkleize_with_floor_scan(
1529:                 db,
1530:                 metadata,
1531:                 staged_updates,
1532:                 Some(prefetched),
1533:                 |floor, tip, limit, out| fill_candidates(&db.bitmap, floor, tip, limit, out),
1534:             )
1535:             .await
1536:     }
1537: 
1538:     /// Resolve the caller's updates on the strategy pool while gathering and reading the
1539:     /// committed prefix of the floor-raise candidates, overlapping the two. Returns the
1540:     /// resolved batch, the staged updates, and the prefetched candidates to seed
1541:     /// [`merkleize_with_floor_scan`](UnmerkleizedBatch::merkleize_with_floor_scan) with.
1542:     ///
1543:     /// `fill_candidates` must be the same candidate source the subsequent floor raise
1544:     /// scans, so the prefetched prefix continues seamlessly into the live scan (see
1545:     /// [`PrefetchedCandidates`]). The gather is clamped to the committed boundary: a
1546:     /// speculative source (e.g. the current variant's parent bitmap) extends past it, but
1547:     /// its candidate sequence below the boundary is identical and only committed locations
1548:     /// are servable by the log read.
1549:     ///
1550:     /// On early exhaustion of the committed set bits, sources may hand back either one past
1551:     /// the last emitted candidate or the committed boundary as the continuation point. Both
1552:     /// are correct: the skipped span holds no set bits, and the source cannot change during
1553:     /// the call (commits and prunes take `&mut` on the database).
1554:     #[allow(clippy::type_complexity)]
1555:     pub(crate) async fn resolve_updates_prefetched<E, C, I, const N: usize>(
1556:         self,
1557:         updates: Vec<(usize, Option<V::Value>)>,
1558:         upserts: Vec<(K, Option<V::Value>)>,
1559:         db: &Db<F, E, C, I, H, update::Unordered<K, V>, N, S>,
1560:         mut fill_candidates: impl FnMut(Location<F>, u64, usize, &mut Vec<Location<F>>) -> Location<F>,
1561:     ) -> Result<
1562:         (
1563:             UnmerkleizedBatch<F, H, update::Unordered<K, V>, S>,
1564:             StagedUpdates<F, update::Unordered<K, V>>,
1565:             PrefetchedCandidates<F, update::Unordered<K, V>>,
1566:         ),
1567:         crate::qmdb::Error<F>,
1568:     >
1569:     where
1570:         E: Context,
1571:         C: Contiguous<Item = Operation<F, update::Unordered<K, V>>>,
1572:         I: UnorderedIndex<Value = Location<F>>,
1573:     {
1574:         // Bound the steps the floor raise can take: only emitted ops consume steps, and an
1575:         // op is emitted per location-resolved update plus per upsert or prior mutation on a
1576:         // key alive in the committed snapshot. Fresh-key creates never consume a step, so
1577:         // unresolved update slots and writes missing from the snapshot are excluded (one
1578:         // in-memory probe per key). The bound is approximate in both directions. Surplus
1579:         // candidates (a translated-key collision, or a key an ancestor already deleted) are
1580:         // dropped by the raise once it moves enough ops, and a shortfall (a write resolving
1581:         // only through an ancestor diff) makes the raise fall back to the live scan when
1582:         // the prefetched prefix runs out.
1583:         let resolved_updates = updates
1584:             .iter()
1585:             .filter(|(slot, _)| self.resolutions.get(*slot).is_some_and(Option::is_some))
1586:             .count()
1587:             .min(self.keys.len());
1588:         let existing_writes = upserts
1589:             .iter()
1590:             .map(|(key, _)| key)
1591:             .chain(self.batch.mutations.keys())
1592:             .filter(|&key| db.snapshot.get(key).next().is_some())
1593:             .count();
1594:         let steps_bound = resolved_updates + existing_writes + 1;
1595: 
1596:         // Overlap the serial update resolution with the candidate prefetch: the
1597:         // committed-prefix candidate set depends only on the base floor, the candidate
1598:         // source, and the step bound, none of which depend on the resolution. The batch
1599:         // moves into the job, so its floor is captured first.
1600:         let scan_from = self.batch.base.inactivity_floor_loc();
1601:         let resolve_len = updates.len() + upserts.len();
1602:         let resolve = db.strategy().spawn(resolve_len, move |strategy| {
1603:             self.resolve_updates(updates, upserts, &strategy)
1604:         });
1605: 
1606:         // Gather the committed-prefix candidates and read their operations, sharded, while
1607:         // the resolution job runs.
1608:         let committed_tip = bitmap::Readable::<N>::len(&*db.bitmap);
1609:         let mut locs: Vec<Location<F>> = Vec::with_capacity(steps_bound);
1610:         let next_scan = fill_candidates(scan_from, committed_tip, steps_bound, &mut locs);
1611:         let raw: Vec<u64> = locs.iter().map(|loc| **loc).collect();
1612:         let read = db.log.read_many_sharded(&raw).await;
1613: 
1614:         // Join the resolution and surface any read failure.
1615:         let (batch, staged_updates) = resolve.await;
1616:         let prefetched = PrefetchedCandidates {
1617:             locs,
1618:             shards: read?,
1619:             next_scan,
1620:         };
1621:         Ok((batch, staged_updates, prefetched))
1622:     }
1623: }

1625: impl<F: Family, K, V, H, S: Strategy> Staged<F, H, update::Ordered<K, V>, S>
1626: where
1627:     K: Key,
1628:     V: ValueEncoding,
1629:     H: Hasher,
1630:     Operation<F, update::Ordered<K, V>>: Codec,
1631: {
1632:     /// Record updates for staged reads and upserts for unread keys, then merkleize.
1633:     ///
1634:     /// Consumes the staged handle and write vectors. Call [`expand`](Staged::expand) before this
1635:     /// method if more keys must be read into the staged index space.
1636:     ///
1637:     /// A `Some` value is an upsert. `None` is a delete. Update indices refer to the staged read
1638:     /// set: the initial [`stage`](UnmerkleizedBatch::stage) input followed by any
1639:     /// [`expand`](Staged::expand) ranges. `metadata` is committed with the returned batch.
1640:     ///
1641:     /// # Panics
1642:     ///
1643:     /// Panics if any update's `read_index` is out of the staged read range.
1644:     #[allow(clippy::type_complexity)]
1645:     #[tracing::instrument(
1646:         name = "qmdb.any.ordered.batch.merkleize.staged",
1647:         level = "info",
1648:         skip_all,
1649:         fields(updates = updates.len() as u64, upserts = upserts.len() as u64),
1650:     )]
1651:     pub async fn merkleize<E, C, I, const N: usize>(
1652:         self,
1653:         updates: Vec<(usize, Option<V::Value>)>,
1654:         upserts: Vec<(K, Option<V::Value>)>,
1655:         metadata: Option<V::Value>,
1656:         db: &Db<F, E, C, I, H, update::Ordered<K, V>, N, S>,
1657:     ) -> Result<Arc<MerkleizedBatch<F, H::Digest, update::Ordered<K, V>, S>>, crate::qmdb::Error<F>>
1658:     where
1659:         E: Context,
1660:         C: Mutable<Item = Operation<F, update::Ordered<K, V>>>,
1661:         I: OrderedIndex<Value = Location<F>>,
1662:     {
1663:         let (batch, staged_updates) = self.resolve_updates(updates, upserts, db.strategy());
1664:         batch
1665:             .merkleize_with_floor_scan(db, metadata, staged_updates, |floor, tip, limit, out| {
1666:                 fill_candidates(&db.bitmap, floor, tip, limit, out)
1667:             })
1668:             .await
1669:     }
1670: }

1672: // Generic get() for both ordered and unordered UnmerkleizedBatch.
1673: impl<F: Family, H, U, S: Strategy> UnmerkleizedBatch<F, H, U, S>
1674: where
1675:     U: update::Update,
1676:     H: Hasher,
1677:     Operation<F, U>: Codec,
1678: {
1679:     /// Return true when reads can bypass uncommitted overlay resolution and go directly to the DB.
1680:     fn reads_committed_only(&self) -> bool {
1681:         self.mutations.is_empty() && self.base.parent().is_none()
1682:     }
1683: 
1684:     /// Resolve keys against this batch's mutations and any live ancestor diffs, returning partial
1685:     /// results and the unresolved slots that still need committed DB reads.
1686:     ///
1687:     /// `on_diff_hit` is invoked with each slot resolved by an ancestor diff entry (slots
1688:     /// resolved by this batch's mutations do not report), so staged reads can record
1689:     /// ancestor resolutions.
1690:     fn resolve_uncommitted_reads<'a>(
1691:         &self,
1692:         keys: &[&'a U::Key],
1693:         strategy: &S,
1694:         on_diff_hit: impl FnMut(usize, &DiffEntry<F, U::Value>),
1695:     ) -> UncommittedReadResolution<'a, U::Key, U::Value>
1696:     where
1697:         U::Value: Send + Sync,
1698:     {
1699:         let ancestors = self.base.parent().map(|parent| {
1700:             let mut ancestors = vec![Arc::clone(parent)];
1701:             ancestors.extend(parent.ancestors());
1702:             ancestors
1703:         });
1704:         let diffs: Vec<_> = ancestors
1705:             .iter()
1706:             .flatten()
1707:             .map(|batch| batch.diff.as_slice())
1708:             .collect();
1709:         resolve_reads(
1710:             keys,
1711:             |key| self.mutations.get(key).cloned(),
1712:             &diffs,
1713:             strategy,
1714:             on_diff_hit,
1715:         )
1716:     }
1717: 
1718:     /// Read unresolved slots from the committed DB and merge them back into `results`.
1719:     async fn fill_committed_reads<E, C, I, T: Send, const N: usize>(
1720:         unresolved: Vec<PendingRead<'_, U::Key>>,
1721:         db: &Db<F, E, C, I, H, U, N, S>,
1722:         results: &mut [Option<U::Value>],
1723:         map: impl Fn(&U, Location<F>) -> T + Send + Sync,
1724:         mut apply: impl FnMut(usize, T) -> U::Value,
1725:     ) -> Result<(), crate::qmdb::Error<F>>
1726:     where
1727:         E: Context,
1728:         C: Contiguous<Item = Operation<F, U>>,
1729:         I: UnorderedIndex<Value = Location<F>> + 'static,
1730:     {
1731:         if unresolved.is_empty() {
1732:             return Ok(());
1733:         }
1734: 
1735:         let db_keys: Vec<_> = unresolved.iter().map(|(_, key)| *key).collect();
1736:         let db_results = db.get_many_map(&db_keys, map).await?;
1737:         for ((slot, _), result) in unresolved.into_iter().zip(db_results) {
1738:             results[slot] = result.map(|value| apply(slot, value));
1739:         }
1740:         Ok(())
1741:     }
1742: 
1743:     /// Read through: mutations -> ancestor diffs -> committed DB.
1744:     pub async fn get<E, C, I, const N: usize>(
1745:         &self,
1746:         key: &U::Key,
1747:         db: &Db<F, E, C, I, H, U, N, S>,
1748:     ) -> Result<Option<U::Value>, crate::qmdb::Error<F>>
1749:     where
1750:         E: Context,
1751:         C: Contiguous<Item = Operation<F, U>>,
1752:         I: UnorderedIndex<Value = Location<F>> + 'static,
1753:     {
1754:         let mut values = self.get_many(&[key], db).await?;
1755:         Ok(values.pop().expect("one result per key"))
1756:     }
1757: 
1758:     /// Batch read multiple keys (mutations -> ancestor diffs -> committed DB).
1759:     ///
1760:     /// Returns results in the same order as the input keys, with `None` for absent or deleted
1761:     /// keys. Resolved locations are not retained, so writing a key read only through `get_many`
1762:     /// requires an index re-probe and journal re-read during merkleize. Use
1763:     /// [`stage`](Self::stage) for keys that may be written. When the writable subset is known and
1764:     /// much smaller than the full read set, call `get_many` for the read-only keys first, then
1765:     /// [`stage`](Self::stage) only the writable keys.
1766:     pub async fn get_many<E, C, I, const N: usize>(
1767:         &self,
1768:         keys: &[&U::Key],
1769:         db: &Db<F, E, C, I, H, U, N, S>,
1770:     ) -> Result<Vec<Option<U::Value>>, crate::qmdb::Error<F>>
1771:     where
1772:         E: Context,
1773:         C: Contiguous<Item = Operation<F, U>>,
1774:         I: UnorderedIndex<Value = Location<F>> + 'static,
1775:     {
1776:         if keys.is_empty() {
1777:             return Ok(Vec::new());
1778:         }
1779:         if self.reads_committed_only() {
1780:             return db.get_many(keys).await;
1781:         }
1782: 
1783:         let (mut results, unresolved) =
1784:             self.resolve_uncommitted_reads(keys, db.strategy(), |_, _| {});
1785:         Self::fill_committed_reads(
1786:             unresolved,
1787:             db,
1788:             &mut results,
1789:             |data, _| data.value().clone(),
1790:             |_, value| value,
1791:         )
1792:         .await?;
1793: 
1794:         Ok(results)
1795:     }
1796: 
1797:     /// Batch read multiple keys and return a staged batch for the same keys.
1798:     ///
1799:     /// Returns results in the same order as the input keys. The staged batch records updates by
1800:     /// read index: the initial keys occupy `0..keys.len()`, and each
1801:     /// [`expand`](Staged::expand) appends another index range. Unlike
1802:     /// [`get_many`](Self::get_many), the resolved locations are reused at merkleize, so keys
1803:     /// that are read and then written skip the index re-probe and journal re-read.
1804:     #[allow(clippy::type_complexity)]
1805:     #[tracing::instrument(
1806:         name = "qmdb.any.batch.stage",
1807:         level = "info",
1808:         skip_all,
1809:         fields(keys = keys.len() as u64),
1810:     )]
1811:     pub async fn stage<E, C, I, const N: usize>(
1812:         self,
1813:         keys: &[&U::Key],
1814:         db: &Db<F, E, C, I, H, U, N, S>,
1815:     ) -> Result<(Vec<Option<U::Value>>, Staged<F, H, U, S>), crate::qmdb::Error<F>>
1816:     where
1817:         E: Context,
1818:         C: Contiguous<Item = Operation<F, U>>,
1819:         I: UnorderedIndex<Value = Location<F>> + 'static,
1820:     {
1821:         let (results, keys, resolutions) = self.stage_reads(keys, db).await?;
1822:         Ok((
1823:             results,
1824:             Staged {
1825:                 batch: self,
1826:                 keys,
1827:                 resolutions,
1828:             },
1829:         ))
1830:     }
1831: 
1832:     /// Read keys through this batch and return the values plus one owned key and resolution per
1833:     /// staged slot. Location-resolved slots (committed, or ancestor-diff when the update kind
1834:     /// stages those) carry the location and cached payload they resolved to.
1835:     #[allow(clippy::type_complexity)]
1836:     async fn stage_reads<E, C, I, const N: usize>(
1837:         &self,
1838:         keys: &[&U::Key],
1839:         db: &Db<F, E, C, I, H, U, N, S>,
1840:     ) -> Result<
1841:         (
1842:             Vec<Option<U::Value>>,
1843:             Vec<U::Key>,
1844:             Vec<StagedResolution<F, U>>,
1845:         ),
1846:         crate::qmdb::Error<F>,
1847:     >
1848:     where
1849:         E: Context,
1850:         C: Contiguous<Item = Operation<F, U>>,
1851:         I: UnorderedIndex<Value = Location<F>> + 'static,
1852:     {
1853:         let mut resolutions: Vec<StagedResolution<F, U>> =
1854:             iter::repeat_with(|| None).take(keys.len()).collect();
1855: 
1856:         // Record ancestor-diff resolutions when the update kind stages them: the staged
1857:         // write then reuses the resolved location at merkleize instead of falling back to a
1858:         // normal mutation (whose cost -- location gathering, a journal re-read, and
1859:         // per-key ancestor re-resolution -- otherwise grows with ancestor overlap).
1860:         let (mut results, unresolved) =
1861:             self.resolve_uncommitted_reads(keys, db.strategy(), |slot, entry| {
1862:                 let Some(cached) = U::STAGES_ANCESTORS else {
1863:                     return;
1864:                 };
1865:                 if let DiffEntry::Active {
1866:                     loc, base_old_loc, ..
1867:                 } = entry
1868:                 {
1869:                     resolutions[slot] = Some((
1870:                         StagedLoc::Ancestor {
1871:                             loc: *loc,
1872:                             base_old_loc: *base_old_loc,
1873:                         },
1874:                         cached,
1875:                     ));
1876:                 }
1877:             });
1878:         Self::fill_committed_reads(
1879:             unresolved,
1880:             db,
1881:             &mut results,
1882:             |data, loc| (data.value().clone(), loc, data.cached()),
1883:             |slot, (value, loc, payload)| {
1884:                 resolutions[slot] = Some((StagedLoc::Committed(loc), payload));
1885:                 value
1886:             },
1887:         )
1888:         .await?;
1889:         Ok((
1890:             results,
1891:             keys.iter().map(|key| (*key).to_owned()).collect(),
1892:             resolutions,
1893:         ))
1894:     }
1895: }

1898: impl<F: Family, K, V, H, S: Strategy> UnmerkleizedBatch<F, H, update::Unordered<K, V>, S>
1899: where
1900:     K: Key,
1901:     V: ValueEncoding,
1902:     H: Hasher,
1903:     Operation<F, update::Unordered<K, V>>: Codec,
1904: {
1905:     /// Resolve mutations into operations, merkleize, and return an `Arc<MerkleizedBatch>`.
1906:     #[allow(clippy::type_complexity)]
1907:     #[tracing::instrument(
1908:         name = "qmdb.any.unordered.batch.merkleize",
1909:         level = "info",
1910:         skip_all,
1911:         fields(mutations = self.mutations.len() as u64),
1912:     )]
1913:     pub async fn merkleize<E, C, I, const N: usize>(
1914:         self,
1915:         db: &Db<F, E, C, I, H, update::Unordered<K, V>, N, S>,
1916:         metadata: Option<V::Value>,
1917:     ) -> Result<Arc<MerkleizedBatch<F, H::Digest, update::Unordered<K, V>, S>>, crate::qmdb::Error<F>>
1918:     where
1919:         E: Context,
1920:         C: Mutable<Item = Operation<F, update::Unordered<K, V>>>,
1921:         I: UnorderedIndex<Value = Location<F>>,
1922:     {
1923:         self.merkleize_with_floor_scan(
1924:             db,
1925:             metadata,
1926:             StagedUpdates::<F, update::Unordered<K, V>>::new(),
1927:             None,
1928:             |floor, tip, limit, out| fill_candidates(&db.bitmap, floor, tip, limit, out),
1929:         )
1930:         .await
1931:     }
1932: 
1933:     /// Like [`merkleize`](Self::merkleize), but consumes staged updates recorded by
1934:     /// [`Staged::merkleize`] (loaded keys skip the journal re-read their resolution would
1935:     /// otherwise require) and accepts the floor-raise candidate source, optionally seeded
1936:     /// with prefetched committed-prefix candidates that must come from the same floor and
1937:     /// the same candidate source the callback scans (see [`PrefetchedCandidates`]).
1938:     ///
1939:     /// The callback must yield candidates in ascending location order, both within one call
1940:     /// and across successive calls (the floor raise asserts this). It may skip locations only
1941:     /// when it knows they are inactive. The floor-raise loop revalidates each returned
1942:     /// candidate against the batch diff, ancestor diffs, and snapshot because the bitmap
1943:     /// reflects committed state only -- uncommitted ancestor ops aren't tracked, and bits can
1944:     /// be set for locations superseded by an overlay in this chain.
1945:     pub(crate) async fn merkleize_with_floor_scan<E, C, I, const N: usize>(
1946:         self,
1947:         db: &Db<F, E, C, I, H, update::Unordered<K, V>, N, S>,
1948:         metadata: Option<V::Value>,
1949:         staged_updates: StagedUpdates<F, update::Unordered<K, V>>,
1950:         prefetched: Option<PrefetchedCandidates<F, update::Unordered<K, V>>>,
1951:         fill_candidates: impl FnMut(Location<F>, u64, usize, &mut Vec<Location<F>>) -> Location<F>,
1952:     ) -> Result<Arc<MerkleizedBatch<F, H::Digest, update::Unordered<K, V>, S>>, crate::qmdb::Error<F>>
1953:     where
1954:         E: Context,
1955:         C: Mutable<Item = Operation<F, update::Unordered<K, V>>>,
1956:         I: UnorderedIndex<Value = Location<F>>,
1957:     {
1958:         let (mut mutations, m) = self.into_parts();
1959: 
1960:         // Resolve existing keys.
1961:         let locations = m.gather_existing_locations(&mutations, db, false);
1962:         let results = m.read_ops(&locations, &[], &db.log).await?;
1963: 
1964:         // Generate user mutation operations.
1965:         let mut ops: Vec<Operation<F, update::Unordered<K, V>>> =
1966:             Vec::with_capacity(mutations.len() + staged_updates.len() + 1);
1967:         let mut diff: DiffVec<K, F, V::Value> =
1968:             Vec::with_capacity(mutations.len() + staged_updates.len());
1969: 
1970:         // Committed locations superseded by this batch, collected for the floor raise (which
1971:         // skips re-reading them). Emission order is ascending in `base_old_loc` except for
1972:         // entries resolved through ancestor diffs, so `finish` usually skips its sort.
1973:         let mut superseded_locs: Vec<Location<F>> = Vec::with_capacity(diff.capacity());
1974:         let mut active_keys_delta: isize = 0;
1975:         let mut user_steps: u64 = 0;
1976: 
1977:         // Write a user mutation at the next batch location, preserving the previous committed
1978:         // location of the key it supersedes.
1979:         let mut emit = |key: K, base_old_loc: Option<Location<F>>, mutation: Option<V::Value>| {
1980:             let new_loc = m.base_state.size + ops.len() as u64;
1981:             superseded_locs.extend(base_old_loc);
1982:             match mutation {
1983:                 Some(value) => {
1984:                     ops.push(Operation::Update(update::Unordered(
1985:                         key.clone(),
1986:                         value.clone(),
1987:                     )));
1988:                     diff.push((
1989:                         key,
1990:                         DiffEntry::Active {
1991:                             value,
1992:                             loc: new_loc,
1993:                             base_old_loc,
1994:                         },
1995:                     ));
1996:                 }
1997:                 None => {
1998:                     ops.push(Operation::Delete(key.clone()));
1999:                     diff.push((key, DiffEntry::Deleted { base_old_loc }));
2000:                     active_keys_delta -= 1;
2001:                 }
2002:             }
2003:             user_steps += 1;
2004:         };
2005: 
2006:         // Process updates/deletes of existing keys in location order, merging staged entries
2007:         // into the read results. This includes keys from both the committed snapshot and ancestor
2008:         // diffs. A staged entry's `value` is `Some` for an update and `None` for a delete, and
2009:         // `emit` writes it as an `Update`/`Delete` at the staged location. An ancestor-staged
2010:         // entry orders by its ancestor location but supersedes the key's committed base
2011:         // location, exactly as its mutation-fallback path would have.
2012:         //
2013:         // A staged location below the merkleize-time committed boundary means the resolving
2014:         // ancestor has committed and dropped out of the alive chain, retiring the recorded
2015:         // base (see [`StagedLoc`]). The location itself is then the committed location this
2016:         // write supersedes, matching what the fallback path's live-snapshot resolution would
2017:         // produce. Resolutions whose ancestor is still alive keep their recorded base. If
2018:         // that ancestor commits before this batch is applied, `apply_batch` resolves the
2019:         // key in the ancestor's traveling diff and supersedes its entry's location instead.
2020:         let staged_base_old_loc = |sloc: StagedLoc<F>| match sloc {
2021:             StagedLoc::Committed(loc) => Some(loc),
2022:             StagedLoc::Ancestor { loc, .. } if *loc < m.db_state.size => Some(loc),
2023:             StagedLoc::Ancestor { base_old_loc, .. } => base_old_loc,
2024:         };
2025:         let mut cached = staged_updates.into_iter().peekable();
2026:         for (op, &old_loc) in results.iter().zip(&locations) {
2027:             while cached
2028:                 .peek()
2029:                 .is_some_and(|&(_, sloc, (), _)| sloc.loc() < old_loc)
2030:             {
2031:                 let (key, sloc, (), mutation) = cached.next().expect("peeked entry exists");
2032:                 emit(key, staged_base_old_loc(sloc), mutation);
2033:             }
2034: 
2035:             let key = op.key().expect("updates should have a key");
2036: 
2037:             // A key resolved via the ancestor diff must only match at its ancestor-diff
2038:             // location. Without this guard, a stale snapshot collision (the pre-parent DB
2039:             // snapshot still containing the key's old location) can consume the mutation at the
2040:             // wrong sort position, changing the operation order relative to the committed-state
2041:             // path. When the ancestor diff entry does match, use it to trace `base_old_loc`
2042:             // back to the key's location in the committed DB snapshot.
2043:             let base_old_loc = if let Some(entry) = resolve_in_ancestors(&m.ancestors, key) {
2044:                 if entry.loc() != Some(old_loc) {
2045:                     continue;
2046:                 }
2047:                 entry.base_old_loc()
2048:             } else {
2049:                 Some(old_loc)
2050:             };
2051: 
2052:             let Some(mutation) = mutations.remove(key) else {
2053:                 // Snapshot index collision: this operation's key does not match
2054:                 // any mutation key. The mutation will be handled as a create below.
2055:                 continue;
2056:             };
2057: 
2058:             emit(key.clone(), base_old_loc, mutation);
2059:         }
2060:         for (key, sloc, (), mutation) in cached {
2061:             emit(key, staged_base_old_loc(sloc), mutation);
2062:         }
2063: 
2064:         // Handle parent-deleted keys that the child wants to re-create.
2065:         let parent_deleted_creates = m.extract_parent_deleted_creates(&mut mutations);
2066: 
2067:         // Process creates: remaining mutations (fresh keys) plus parent-deleted

2600:     ///
2601:     /// All uncommitted ancestors in the chain must be kept alive until the child (or any
2602:     /// descendant) is merkleized. Dropping an uncommitted ancestor causes data

2625:     /// Inclusion proof for the operations returned by [`Self::operations`], anchored at
2626:     /// this batch's tip. The pair verifies against [`Self::root`] via
2627:     /// [`crate::qmdb::verify_proof`]. Together with [`Self::pinned_nodes`] they verify via
2628:     /// [`crate::qmdb::verify_proof_and_pinned_nodes`].
2629:     ///
2630:     /// Nodes of unapplied ancestors are read through the chain, so those ancestors must still be
2631:     /// alive. Nodes below the chain are read from `db`'s
2632:     /// [Merkle store][crate::merkle::mem::Mem], which retains them at least until
2633:     /// this batch's changes are flushed (by a commit or sync after apply).

2685:     /// Read through: local diff -> parent chain -> committed DB.
2686:     pub async fn get<E, C, I, H, const N: usize>(
2687:         &self,
2688:         key: &U::Key,
2689:         db: &Db<F, E, C, I, H, U, N, S>,
2690:     ) -> Result<Option<U::Value>, crate::qmdb::Error<F>>
2691:     where
2692:         E: Context,
2693:         C: Contiguous<Item = Operation<F, U>>,
2694:         I: UnorderedIndex<Value = Location<F>> + 'static,
2695:         H: Hasher<Digest = D>,
2696:     {
2697:         if let Some(entry) = lookup_sorted(self.diff.as_slice(), key) {
2698:             return Ok(entry.value().cloned());
2699:         }
2700:         // Walk parent chain. If a parent was freed (committed and dropped), the iterator
2701:         // stops and we fall through to DB.
2702:         for batch in self.ancestors() {
2703:             if let Some(entry) = lookup_sorted(batch.diff.as_slice(), key) {
2704:                 return Ok(entry.value().cloned());
2705:             }
2706:         }
2707:         db.get(key).await
2708:     }

2710:     /// Batch read multiple keys.
2711:     ///
2712:     /// Returns results in the same order as the input keys.
2713:     pub async fn get_many<E, C, I, H, const N: usize>(
2714:         &self,
2715:         keys: &[&U::Key],
2716:         db: &Db<F, E, C, I, H, U, N, S>,
2717:     ) -> Result<Vec<Option<U::Value>>, crate::qmdb::Error<F>>
2718:     where
2719:         E: Context,
2720:         C: Contiguous<Item = Operation<F, U>>,
2721:         I: UnorderedIndex<Value = Location<F>> + 'static,
2722:         H: Hasher<Digest = D>,
2723:     {
2724:         if keys.is_empty() {
2725:             return Ok(Vec::new());
2726:         }
2727: 
2728:         let ancestors: Vec<_> = self.ancestors().collect();
2729:         let diffs: Vec<_> = ancestors
2730:             .iter()
2731:             .map(|batch| batch.diff.as_slice())
2732:             .collect();
2733:         let (mut results, unresolved) = resolve_reads(
2734:             keys,
2735:             |key| lookup_sorted(self.diff.as_slice(), key).map(|entry| entry.value().cloned()),
2736:             &diffs,
2737:             db.strategy(),
2738:             |_, _| {},
2739:         );
2740: 
2741:         if !unresolved.is_empty() {
2742:             let db_keys: Vec<_> = unresolved.iter().map(|(_, key)| *key).collect();
2743:             let db_results = db.get_many(&db_keys).await?;
2744:             for ((slot, _), value) in unresolved.into_iter().zip(db_results) {
2745:                 results[slot] = value;
2746:             }
2747:         }
2748: 
2749:         Ok(results)
2750:     }

2789: where
2790:     F: Family,
2791:     E: Context,
2792:     C: Mutable<Item = Operation<F, U>>,
2793:     I: UnorderedIndex<Value = Location<F>>,
2794:     H: Hasher,
2795:     U: update::Update,
2796:     S: Strategy,
2797:     Operation<F, U>: Codec,
2798: {
2799:     /// Check that `batch` can be applied to the database in its current state, without
2800:     /// applying it.
2801:     ///
2802:     /// [`Self::apply_batch`] runs the same validation but consumes the database when it
2803:     /// fails; callers that want to reject a bad batch and keep the handle can check first.
2804:     pub fn validate_batch(
2805:         &self,
2806:         batch: &MerkleizedBatch<F, H::Digest, U, S>,
2807:     ) -> Result<(), crate::qmdb::Error<F>> {
2808:         batch
2809:             .bounds
2810:             .validate_apply_to(self.commitment(), self.inactivity_floor_loc)
2811:     }
2812: 
2813:     /// Apply a batch to the database, returning the range of written operations.
2814:     ///
2815:     /// A batch is valid only if every batch applied to the database since this batch's
2816:     /// ancestor chain was created is an ancestor of this batch. Applying a batch from a
2817:     /// different fork returns [`crate::qmdb::Error::StaleBatch`] (see
2818:     /// [`crate::qmdb::batch_chain`] for more details).
2819:     ///
2820:     /// This publishes the batch to the in-memory database state and appends it to the journal.
2821:     /// Call [`Db::commit`] or [`Db::sync`], or await the handle returned by [`Db::start_sync`], to
2822:     /// make the applied state durable.
2823:     #[tracing::instrument(
2824:         name = "qmdb.any.db.apply_batch",
2825:         level = "info",
2826:         skip_all,
2827:         fields(
2828:             batch_total_size = *batch.bounds.tip.size,
2829:             batch_base_size = *batch.bounds.base.size,
2830:             db_size = *self.last_commit_loc + 1,
2831:             ancestor_batches = batch.ancestor_diffs.len() as u64,
2832:         ),
2833:     )]
2834:     pub async fn apply_batch(
2835:         mut self,
2836:         batch: Arc<MerkleizedBatch<F, H::Digest, U, S>>,
2837:     ) -> Result<(Self, Range<Location<F>>), crate::qmdb::Error<F>> {
2838:         let _timer = self.metrics.apply_batch_timer();
2839:         self.metrics.apply_batch_calls.inc();
2840:         self.validate_batch(&batch)?;
2841:         let db_size = *self.last_commit_loc + 1;
2842:         let start_loc = Location::new(db_size);
2843: 
2844:         // Apply journal (handles its own partial ancestor skipping).
2845:         self.log = self.log.apply_batch(&batch.journal_batch).await?;
2846: 
2847:         // Scoped so the bitmap guard drops before later `.await`s (guard is `!Send`).
2848:         {
2849:             let mut bitmap = self.bitmap.write();
2850:             bitmap.extend_to(*batch.bounds.tip.size);
2851: 
2852:             if batch.ancestor_diffs.is_empty() {
2853:                 // Fast path: no ancestors to merge, no fixups to look up.
2854:                 for (key, entry) in batch.diff.iter() {
2855:                     apply_diff(
2856:                         &mut self.snapshot,
2857:                         &mut bitmap,
2858:                         key,
2859:                         entry,
2860:                         entry.base_old_loc(),
2861:                     );
2862:                 }
2863:             } else {
2864:                 // Partition ancestor diffs into already-applied (provide `base_old_loc` fixups)
2865:                 // and pending (still to be applied; merged with the child).
2866:                 let mut applied = Vec::with_capacity(batch.ancestor_diffs.len());
2867:                 let mut pending = Vec::with_capacity(batch.ancestor_diffs.len());
2868:                 for (i, ancestor_diff) in batch.ancestor_diffs.iter().enumerate() {
2869:                     if batch.bounds.ancestors[i].state.size <= db_size {
2870:                         applied.push(ancestor_diff.as_slice());
2871:                     } else {
2872:                         pending.push(ancestor_diff.as_slice());
2873:                     }
2874:                 }
2875:                 let mut resolver = DiffCursors::new(applied);
2876:                 if batch.ancestor_base_locs.is_empty() {
2877:                     let merge = DiffMerge::new(
2878:                         iter::once(batch.diff.as_slice()).chain(pending.iter().copied()),
2879:                     );
2880:                     for (key, entry) in merge {
2881:                         let old = resolver
2882:                             .resolve(key)
2883:                             .map(DiffEntry::loc)
2884:                             .unwrap_or_else(|| entry.base_old_loc());
2885:                         apply_diff(&mut self.snapshot, &mut bitmap, key, entry, old);
2886:                     }
2887:                 } else {
2888:                     let mut ancestor_base_locs = batch.ancestor_base_locs.iter().peekable();
2889:                     let merge = DiffMerge::new(
2890:                         iter::once(batch.diff.as_slice()).chain(pending.iter().copied()),
2891:                     );
2892:                     for (key, entry) in merge {
2893:                         let old = resolver.resolve(key).map_or_else(
2894:                             || {
2895:                                 while ancestor_base_locs
2896:                                     .peek()
2897:                                     .is_some_and(|(candidate, _)| candidate < key)
2898:                                 {
2899:                                     ancestor_base_locs.next();
2900:                                 }
2901:                                 if ancestor_base_locs
2902:                                     .peek()
2903:                                     .is_some_and(|(candidate, _)| candidate == key)
2904:                                 {
2905:                                     ancestor_base_locs.next().expect("peeked entry exists").1
2906:                                 } else {
2907:                                     entry.base_old_loc()
2908:                                 }
2909:                             },
2910:                             DiffEntry::loc,
2911:                         );
2912:                         apply_diff(&mut self.snapshot, &mut bitmap, key, entry, old);
2913:                     }
2914:                 }
2915:             }
2916: 
2917:             // CommitFloor: bit = 1 only on the current last commit. Demote the previous and
2918:             // set the new; earlier ancestor commits between them are already 0 from
2919:             // `extend_to`.
2920:             bitmap.set_bit(*self.last_commit_loc, false);
2921:             bitmap.set_bit(*batch.bounds.tip.size - 1, true);
2922:         }
2923: 
2924:         // Update DB metadata.
2925:         self.active_keys = batch.total_active_keys;
2926:         self.inactivity_floor_loc = batch.bounds.inactivity_floor;
2927:         self.last_commit_loc = batch.bounds.tip.size - 1;
2928:         self.root = batch.root();
2929: 
2930:         // Return range of operations that were written to the log.
2931:         let end_loc = self.last_commit_loc + 1;
2932:         let range = start_loc..end_loc;
2933:         self.update_metrics();
2934:         self.metrics
2935:             .operations_applied
2936:             .inc_by(*range.end - *range.start);
2937:         Ok((self, range))
2938:     }
2939: }

3306:     /// Single-step oracle for [`fill_candidates`]: return the next floor-raise candidate in
3307:     /// `[floor, tip)`. `bitmap_fill_candidates_matches_oracle` proves the production batch
3308:     /// fill produces this exact sequence.

3310:         bitmap: &Shared<N>,
3311:         floor: Location<F>,
3312:         tip: u64,
3313:     ) -> Option<Location<F>> {

3315:         let bitmap_len = bitmap::Readable::<N>::len(bitmap);

3324:         (candidate < tip).then(|| Location::new(candidate))
3325:     }
