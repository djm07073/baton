65: /// Unified trait for an authenticated database.
66: ///
67: /// This trait provides access to authentication (root), pruning, persistence,
68: /// reads, and batch mutations.
69: pub trait DbAny<F: Family>:
70:     BatchableDb<Family = F, K = <Self as DbAny<F>>::Key, V = <Self as DbAny<F>>::Value> + Send + Sync
71: {
72:     /// The key type used to look up values.
73:     type Key: Key;
74: 
75:     /// The value type stored in the database.
76:     type Value: CodecShared + Clone;
77: 
78:     /// The digest type used for merkleization.
79:     type Digest: Digest;
80: 
81:     /// Get the value of a key.
82:     fn get<'a>(
83:         &'a self,
84:         key: &'a Self::Key,
85:     ) -> impl Future<Output = Result<Option<Self::Value>, Error<F>>> + Send + use<'a, F, Self>;
86: 
87:     /// Get the values of multiple keys, returned in the same order as the input keys.
88:     fn get_many<'a>(
89:         &'a self,
90:         keys: &'a [&'a Self::Key],
91:     ) -> impl Future<Output = Result<Vec<Option<Self::Value>>, Error<F>>> + Send + use<'a, F, Self>;
92: 
93:     /// Returns the root digest of the authenticated store.
94:     fn root(&self) -> Self::Digest;
95: 
96:     /// Return [start, end) where `start` and `end - 1` are the Locations of the oldest and newest
97:     /// retained operations respectively.
98:     fn bounds(&self) -> Range<Location<F>>;
99: 
100:     /// Return the Location of the next operation appended to this db.
101:     fn size(&self) -> Location<F> {
102:         self.bounds().end
103:     }
104: 
105:     /// Get the metadata associated with the last commit.
106:     fn get_metadata(
107:         &self,
108:     ) -> impl Future<Output = Result<Option<<Self as DbAny<F>>::Value>, Error<F>>> + Send;
109: 
110:     /// Prune historical operations prior to `loc`.
111:     fn prune(self, loc: Location<F>) -> impl Future<Output = Result<Self, Error<F>>> + Send;
112: 
113:     /// Begin durably persisting the database.
114:     ///
115:     /// Awaiting the returned [Handle] provides the same durability guarantee as [Self::commit]
116:     /// for the state applied before the call, plus a best-effort attempt to bound the recovery
117:     /// needed on startup. Use [Self::sync] to guarantee none is needed.
118:     fn start_sync(self) -> impl Future<Output = Result<(Self, Handle<()>), Error<F>>> + Send;
119: 
120:     /// Durably persist the database, guaranteeing the current state will survive a crash.
121:     ///
122:     /// For a stronger guarantee that eliminates potential recovery, use [Self::sync] instead.
123:     fn commit(self) -> impl Future<Output = Result<Self, Error<F>>> + Send;
124: 
125:     /// Durably persist the database, guaranteeing the current state will survive a crash, and that
126:     /// no recovery will be needed on startup.
127:     ///
128:     /// This provides a stronger guarantee than [Self::commit] but may be slower.
129:     fn sync(self) -> impl Future<Output = Result<Self, Error<F>>> + Send;
130: 
131:     /// Destroy the database, removing all data from disk.
132:     fn destroy(self) -> impl Future<Output = Result<(), Error<F>>> + Send
133:     where
134:         Self: Sized;
135: 
136:     /// The location before which all operations can be pruned.
137:     fn inactivity_floor_loc(&self) -> Location<F>;
138: 
139:     /// The maximum location that [`Self::prune`] accepts and the most recent location from which
140:     /// this database can be safely synced.
141:     fn sync_boundary(&self) -> Location<F>;
142: }

144: /// Proof generation for Any database variants.
145: ///
146: /// Only Any variants implement this trait. Current variants have a different
147: /// proof structure (grafted MMR + activity bitmap) that is incompatible with
148: /// the ops-level proofs returned here.
149: pub trait Provable<F: Family>: DbAny<F> {
150:     /// The operation type stored in the log.
151:     type Operation;
152: 
153:     /// Generate a proof of operations starting at `start_loc`.
154:     #[allow(clippy::type_complexity)]
155:     fn proof(
156:         &self,
157:         start_loc: Location<F>,
158:         max_ops: NonZeroU64,
159:     ) -> impl Future<Output = Result<(Proof<F, Self::Digest>, Vec<Self::Operation>), Error<F>>> + Send
160:     {
161:         async move {
162:             self.historical_proof(self.bounds().end, start_loc, max_ops)
163:                 .await
164:         }
165:     }
166: 
167:     /// Generate a proof of operations starting at `start_loc` for a historical size.
168:     #[allow(clippy::type_complexity)]
169:     fn historical_proof(
170:         &self,
171:         historical_size: Location<F>,
172:         start_loc: Location<F>,
173:         max_ops: NonZeroU64,
174:     ) -> impl Future<Output = Result<(Proof<F, Self::Digest>, Vec<Self::Operation>), Error<F>>> + Send;
175: }
