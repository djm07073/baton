148:     /// Main event loop: multiplexes mailbox messages and engine callbacks.
149:     async fn run(
150:         mut self,
151:         (sender, receiver): (impl Sender<PublicKey = P>, impl Receiver<PublicKey = P>),
152:     ) {
153:         let (handler_tx, mut handler_rx) =
154:             actor_mailbox::new(self.context.child("handler"), self.config.mailbox_size);
155:         let handler = handler::Handler::new(handler_tx);
156:         let (engine, mut resolver_mailbox) = p2p::Engine::new(
157:             self.context.as_present().child("resolver"),
158:             p2p::Config {
159:                 peer_provider: self.config.peer_provider.clone(),
160:                 blocker: self.config.blocker.clone(),
161:                 consumer: handler.clone(),
162:                 producer: handler,
163:                 mailbox_size: self.config.mailbox_size,
164:                 me: self.config.me.clone(),
165:                 timeout: self.config.timeout,
166:                 fetch_retry_timeout: self.config.fetch_retry_timeout,
167:                 priority_requests: self.config.priority_requests,
168:                 priority_responses: self.config.priority_responses,
169:             },
170:         );
171:         let mut resolver_task = engine.start((sender, receiver));
172: 
173:         select_loop! {
174:             self.context,
175:             on_start => {
176:                 self.pending.retain(|_, subs| {
177:                     subs.retain(|s| !s.is_closed());
178:                     !subs.is_empty()
179:                 });
180:                 let mailbox_message = async {
181:                     match self.mailbox_rx.recv().await {
182:                         Some(message) => Some(message),
183:                         None => future::pending().await,
184:                     }
185:                 };
186:             },
187:             on_stopped => {
188:                 return;
189:             },
190:             _ = &mut resolver_task => {
191:                 return;
192:             },
193:             Some(message) = mailbox_message else continue => {
194:                 match self.handle_mailbox_message(message) {
195:                     MailboxAction::None => {}
196:                     MailboxAction::Fetch(request) => {
197:                         resolver_mailbox.fetch(request);
198:                     }
199:                     MailboxAction::Cancel(request) => {
200:                         resolver_mailbox.retain(move |key, _| key != &request);
201:                     }
202:                 }
203:             },
204:             Some(message) = handler_rx.recv() else {
205:                 return;
206:             } => match message {
207:                 handler::EngineMessage::Deliver {
208:                     key,
209:                     value,
210:                     response,
211:                 } => {
212:                     self.handle_deliver(key, value, response).await;
213:                 }
214:                 handler::EngineMessage::Produce { key, response } => {
215:                     self.handle_produce(key, response).await;
216:                 }
217:             },
218:         }
219:     }
220: 
221:     /// Process a mailbox message. Returns a request to fetch if a new key was registered.
222:     fn handle_mailbox_message(&mut self, message: SyncMessage<F, DB>) -> MailboxAction<F> {
223:         match message {
224:             mailbox::Message::AttachDatabase(db) => {
225:                 let replacing_existing = matches!(self.state, State::HasDb(_));
226:                 info!(replacing_existing, "attached resolver database");
227:                 self.state = State::HasDb(db);
228:                 let _ = self.metrics.has_database.try_set(1i64);
229:                 MailboxAction::None
230:             }
231:             mailbox::Message::GetOperations { request, response } => {
232:                 if let Some(subscribers) = self.pending.get_mut(&request) {
233:                     subscribers.retain(|subscriber| !subscriber.is_closed());
234:                     if !subscribers.is_empty() {
235:                         subscribers.push(response);
236:                         return MailboxAction::None;
237:                     }
238:                 }
239:                 self.pending.insert(request, vec![response]);
240:                 self.metrics.fetch_requests.inc();
241:                 let _ = self.metrics.pending_requests.try_set(self.pending.len());
242:                 MailboxAction::Fetch(request)
243:             }
244:             mailbox::Message::CancelOperations { request } => {
245:                 if self.should_cancel_request(&request) {
246:                     self.metrics.cancel_requests.inc();
247:                     let _ = self.metrics.pending_requests.try_set(self.pending.len());
248:                     MailboxAction::Cancel(request)
249:                 } else {
250:                     MailboxAction::None
251:                 }
252:             }
253:         }
254:     }
255: 
256:     /// Returns `true` if a request should be cancelled.
257:     fn should_cancel_request(&mut self, request: &Request<F>) -> bool {
258:         let Some(subscribers) = self.pending.get_mut(request) else {
259:             return true;
260:         };
261:         subscribers.retain(|subscriber| !subscriber.is_closed());
262:         if !subscribers.is_empty() {
263:             return false;
264:         }
265:         self.pending.remove(request);
266:         true
267:     }

269:     /// Decode a peer's response, fan it out to pending subscribers, and aggregate approvals.
270:     async fn handle_deliver(
271:         &mut self,
272:         key: Request<F>,
273:         value: bytes::Bytes,
274:         feedback_tx: oneshot::Sender<bool>,
275:     ) {
276:         // Only accept responses for keys we currently have in-flight.
277:         // Unknown keys are unsolicited/stale deliveries and are ignored.
278:         let Some(subscribers) = self.pending.remove(&key) else {
279:             self.metrics.deliveries.inc(status::Status::Dropped);
280:             feedback_tx.send_lossy(true);
281:             return;
282:         };
283:         let _ = self.metrics.pending_requests.try_set(self.pending.len());
284: 
285:         let cfg = (key.max_ops().get() as usize, ());
286:         let response = match Response::<F, Op<DB>, DatabaseRoot<DB>>::decode_cfg(value, &cfg) {
287:             Ok(response)
288:                 if matches!(
289:                     (&key, &response),
290:                     (Request::Operations { .. }, Response::Operations { .. })
291:                         | (Request::Boundary { .. }, Response::Boundary { .. })
292:                 ) =>
293:             {
294:                 response
295:             }
296:             _ => {
297:                 self.pending.insert(key, subscribers);
298:                 let _ = self.metrics.pending_requests.try_set(self.pending.len());
299:                 self.metrics.deliveries.inc(status::Status::Invalid);
300:                 feedback_tx.send_lossy(false);
301:                 return;
302:             }
303:         };
304: 
305:         let mut approvals = Vec::new();
306:         for subscriber in subscribers {
307:             let (success_tx, success_rx) = oneshot::channel();
308:             if subscriber
309:                 .send((response.clone(), Some(success_tx)))
310:                 .is_err()
311:             {
312:                 continue;
313:             }
314:             approvals.push(success_rx);
315:         }
316: 
317:         if approvals.is_empty() {
318:             self.metrics.deliveries.inc(status::Status::Success);
319:             feedback_tx.send_lossy(true);
320:             return;
321:         }
322: 
323:         let mut peer_valid = true;
324:         for approval in approvals {
325:             if let Ok(approved) = approval.await {
326:                 peer_valid &= approved;
327:             }
328:         }
329: 
330:         if peer_valid {
331:             self.metrics.deliveries.inc(status::Status::Success);
332:         } else {
333:             self.metrics.deliveries.inc(status::Status::Failure);
334:             debug!(?key, "downstream marked response as peer-invalid");
335:         }
336:         feedback_tx.send_lossy(peer_valid);
337:     }
338: 
339:     /// Serve a peer's request by querying the local database.
340:     async fn handle_produce(
341:         &mut self,
342:         key: Request<F>,
343:         response_tx: oneshot::Sender<bytes::Bytes>,
344:     ) {
345:         let State::HasDb(database) = &self.state else {
346:             self.metrics.serve_requests.inc(status::Status::Dropped);
347:             return;
348:         };
349:         if let Request::Operations { max_ops, .. } = key
350:             && max_ops > self.config.max_serve_ops
351:         {
352:             self.metrics.serve_requests.inc(status::Status::Dropped);
353:             return;
354:         }
355:         let result = database.serve(key).await;
356: 
357:         let Ok((response, _feedback_tx)) = result else {
358:             self.metrics.serve_requests.inc(status::Status::Failure);
359:             return;
360:         };
361: 
362:         response_tx.send_lossy(response.encode());
363:         self.metrics.serve_requests.inc(status::Status::Success);
364:     }
365: }
