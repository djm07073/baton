63: /// Wait for the next synchronization event.
64: /// Returns `None` when there are no outstanding requests and no channels to wait on.
65: async fn wait_for_event<F: Family, Op: Send, D: Digest, E: Send>(
66:     update_rx: &mut Option<mpsc::Receiver<Target<F, D>>>,
67:     finish_rx: &mut Option<mpsc::Receiver<()>>,
68:     outstanding_requests: &mut Requests<F, Op, D, E>,
69: ) -> Option<Event<F, Op, D, E>> {
70:     if outstanding_requests.len() == 0 && update_rx.is_none() && finish_rx.is_none() {
71:         return None;
72:     }
73: 
74:     let target_update_fut = update_rx.as_mut().map_or_else(
75:         || Either::Right(pending()),
76:         |update_rx| Either::Left(update_rx.recv()),
77:     );
78:     let finish_fut = finish_rx.as_mut().map_or_else(
79:         || Either::Right(pending()),
80:         |finish_rx| Either::Left(finish_rx.recv()),
81:     );
82:     let batch_result_fut = outstanding_requests.next_completed();
83: 
84:     select! {
85:         finish = finish_fut => finish.map_or_else(
86:             || Some(Event::FinishChannelClosed),
87:             |_| Some(Event::FinishRequested)
88:         ),
89:         target = target_update_fut => target.map_or_else(
90:             || Some(Event::UpdateChannelClosed),
91:             |target| Some(Event::TargetUpdate(target))
92:         ),
93:         result = batch_result_fut => Some(Event::BatchReceived(result)),
94:     }
95: }

244:     pub async fn new(config: Config<DB, S>) -> Result<Self, Error<DB, S>> {
245:         if !config.target.range.end().is_valid() {
246:             return Err(SyncError::Engine(EngineError::InvalidTarget {
247:                 lower_bound_pos: config.target.range.start(),
248:                 upper_bound_pos: config.target.range.end(),
249:             }));
250:         }
251: 
252:         // Create journal and verifier using the database's factory methods
253:         let journal = <DB::Journal as Journal<DB::Family>>::new(
254:             config.context.child("journal"),
255:             config.db_config.journal_config(),
256:             config.target.range.clone(),
257:         )
258:         .await?;
259:         let journal_size = journal.size();
260: 
261:         // The sync journal is the source of truth for resume. If it already
262:         // reaches the target, try to recover the target's pinned nodes from local
263:         // Merkle state before asking peers for them. Partial journals resume without
264:         // probing completed database state.
265:         let pinned_nodes = if journal_size == *config.target.range.end() {
266:             DB::local_pinned_nodes(
267:                 config.context.child("local_pinned_nodes"),
268:                 &config.db_config,
269:                 &config.target,
270:                 &journal,
271:             )
272:             .await?
273:         } else {
274:             None
275:         };
276: 
277:         let sync_context = config.context.child("sync");
278:         let metrics = Metrics::new(&sync_context);
279:         let mut engine = Self {
280:             outstanding_requests: Requests::new(),
281:             fetched_operations: BTreeMap::new(),
282:             pinned_nodes,
283:             retained_roots: BTreeMap::new(),
284:             max_retained_roots: config.max_retained_roots,
285:             target: config.target.clone(),
286:             max_outstanding_requests: config.max_outstanding_requests,
287:             fetch_batch_size: config.fetch_batch_size,
288:             apply_batch_size: config.apply_batch_size,
289:             journal,
290:             source: Arc::new(config.source),
291:             hasher: qmdb::hasher::<DB::Hasher>(),
292:             context: config.context,
293:             config: config.db_config,
294:             update_rx: config.update_rx,
295:             finish_requested: false,
296:             finish_rx: config.finish_rx,
297:             reached_target_tx: config.reached_target_tx,
298:             reached_current_target_reported: false,
299:             metrics,
300:         };
301:         engine.schedule_requests()?;
302:         engine.record_progress();
303:         Ok(engine)

306:     /// Track `request` and spawn its fetch against the shared source.
307:     fn spawn_fetch(&mut self, request: Request<DB::Family>) {
308:         let source = Arc::clone(&self.source);
309:         self.outstanding_requests
310:             .insert(request, move |id| async move {
311:                 let result = source.serve(request).await;
312:                 IndexedFetchResult { id, result }
313:             });
314:     }

316:     /// Schedule new fetch requests for operations in the sync range that we haven't yet fetched.
317:     fn schedule_requests(&mut self) -> Result<(), Error<DB, S>> {
318:         let target_size = self.target.range.end();
319: 
320:         // Schedule a boundary request at the lower sync bound if pinned nodes are still
321:         // needed and one isn't already in flight. The pinned nodes it returns are what let
322:         // us rebuild the pruned prefix.
323:         if !self.pinned_nodes_ready()
324:             && !self
325:                 .outstanding_requests
326:                 .contains(&self.target.range.start())
327:         {
328:             let request = Request::Boundary {
329:                 size: target_size,
330:                 start: self.target.range.start(),
331:             };
332:             self.spawn_fetch(request);
333:         }
334: 
335:         // Calculate the maximum number of requests to make
336:         let num_requests = self
337:             .max_outstanding_requests
338:             .saturating_sub(self.outstanding_requests.len());
339: 
340:         let log_size = self.journal.size();
341: 
342:         for _ in 0..num_requests {
343:             // Convert fetched operations to operation counts for shared gap detection
344:             let operation_counts: BTreeMap<Location<DB::Family>, u64> = self
345:                 .fetched_operations
346:                 .iter()
347:                 .map(|(&start_loc, operations)| (start_loc, operations.len() as u64))
348:                 .collect();
349: 
350:             // Find the next gap in the sync range that needs to be fetched.
351:             let Some(gap_range) = crate::qmdb::sync::gaps::find_next(
352:                 Location::new(log_size)..self.target.range.end(),
353:                 &operation_counts,
354:                 self.outstanding_requests.ranges(),
355:             ) else {
356:                 break; // No more gaps to fill
357:             };
358: 
359:             // Calculate batch size for this gap
360:             let gap_size = *gap_range.end.checked_sub(*gap_range.start).unwrap();
361:             let gap_size: NonZeroU64 = gap_size.try_into().unwrap();
362:             let batch_size = self.fetch_batch_size.min(gap_size);
363: 
364:             // Schedule the request
365:             let request = Request::Operations {
366:                 size: target_size,
367:                 start: gap_range.start,
368:                 max_ops: batch_size,
369:             };
370:             self.spawn_fetch(request);
371:         }
372: 
373:         Ok(())
374:     }

376:     /// Reset sync state for a target update.
377:     ///
378:     /// Only cancels requests that cover ranges before the new target range
379:     /// start. Requests at or after the new start are retained; their proofs
380:     /// will be verified against the saved historical root (see
381:     /// `retained_roots`) so the fetched operations can still be used.
382:     pub async fn reset_for_target_update(
383:         mut self,
384:         new_target: Target<DB::Family, DB::Digest>,
385:     ) -> Result<Self, Error<DB, S>> {
386:         self.journal = self.journal.resize(new_target.range.start()).await?;
387:         // Remove requests at or before the new start. The request at start
388:         // must be re-issued as a boundary request with the new target size.
389:         self.outstanding_requests
390:             .remove_before(new_target.range.start().checked_add(1).unwrap());
391:         self.fetched_operations.clear();
392:         self.pinned_nodes = None;
393: 
394:         // Save the current root keyed by its database size for verifying
395:         // retained requests that were issued against this target.
396:         if self.max_retained_roots > 0 {
397:             self.retained_roots
398:                 .insert(self.target.range.end(), self.target.root);
399:             while self.retained_roots.len() > self.max_retained_roots {
400:                 self.retained_roots.pop_first();
401:             }
402:         }
403: 
404:         self.target = new_target;
405:         self.reached_current_target_reported = false;
406:         Ok(self)
407:     }

409:     /// Drain a pending explicit-finish signal without blocking.
410:     ///
411:     /// If a finish signal is present, the finish channel is dropped and the engine
412:     /// may complete as soon as it is at a target. If the finish channel is
413:     /// disconnected before a finish request is observed, this returns
414:     /// [`EngineError::FinishChannelClosed`].
415:     fn drain_finish_requests(&mut self) -> Result<(), Error<DB, S>> {
416:         let Some(finish_rx) = self.finish_rx.as_mut() else {
417:             return Ok(());
418:         };
419:         match finish_rx.try_recv() {
420:             Ok(()) => {
421:                 self.finish_rx = None;
422:                 self.finish_requested = true;
423:                 Ok(())
424:             }
425:             Err(TryRecvError::Empty) => Ok(()),
426:             Err(TryRecvError::Disconnected) => {
427:                 Err(SyncError::Engine(EngineError::FinishChannelClosed))
428:             }
429:         }
430:     }

432:     /// Notify an observer that the current target has been reached. The notification is sent
433:     /// at most once per target, guarded by `reached_current_target_reported`.
434:     ///
435:     /// This send awaits backpressure. When `reached_target_tx` is `Some(...)`,
436:     /// the receiver is expected to consume notifications promptly so the engine
437:     /// can keep making progress. If the receiver side is closed, we drop the
438:     /// sender and continue syncing without further reached-target notifications.
439:     async fn report_reached_target(&mut self) {
440:         if self.reached_current_target_reported {
441:             return;
442:         }
443:         if let Some(sender) = self.reached_target_tx.as_ref()
444:             && !sender.send_lossy(self.target.clone()).await
445:         {
446:             self.reached_target_tx = None;
447:         }
448:         self.reached_current_target_reported = true;
449:     }

457:     /// Store a batch of fetched operations. If the input list is empty, this is a no-op.
458:     pub(crate) fn store_operations(

469:     /// Apply fetched operations to the journal if we have them.
470:     ///
471:     /// This method finds operations that are contiguous with the current journal tip
472:     /// and applies them in order. It removes stale batches and handles partial
473:     /// application of batches when needed.
474:     pub(crate) async fn apply_operations(mut self) -> Result<Self, Error<DB, S>> {
475:         let mut next_loc = self.journal.size();
476: 
477:         // Remove any batches of operations with stale data.
478:         // That is, those whose last operation is before `next_loc`.
479:         self.fetched_operations.retain(|&start_loc, operations| {
480:             assert!(!operations.is_empty());
481:             let end_loc = start_loc.checked_add(operations.len() as u64 - 1).unwrap();
482:             end_loc >= next_loc
483:         });
484: 
485:         loop {
486:             // See if we have the next operation to apply (i.e. at the journal tip).
487:             // Find the index of the range that contains the next location.
488:             let range_start_loc =
489:                 self.fetched_operations
490:                     .iter()
491:                     .find_map(|(range_start, range_ops)| {
492:                         assert!(!range_ops.is_empty());
493:                         let range_end =
494:                             range_start.checked_add(range_ops.len() as u64 - 1).unwrap();
495:                         if *range_start <= next_loc && next_loc <= range_end {
496:                             Some(*range_start)
497:                         } else {
498:                             None
499:                         }
500:                     });
501: 
502:             let Some(range_start_loc) = range_start_loc else {
503:                 // We don't have the next operation to apply (i.e. at the journal tip)
504:                 break;
505:             };
506: 
507:             // Remove the batch of operations that contains the next operation to apply.
508:             let operations = self.fetched_operations.remove(&range_start_loc).unwrap();
509:             assert!(!operations.is_empty());
510:             // Skip operations that are before the next location. The containment check when
511:             // selecting the range (`next_loc <= range_end`) guarantees at least one operation
512:             // at or after it, so the batch is never empty.
513:             let operations = &operations[(next_loc - *range_start_loc) as usize..];
514:             next_loc += operations.len() as u64;
515:             self.journal = self.journal.append(operations).await?;
516:         }
517: 
518:         Ok(self)
519:     }

568:     /// Handle the result of a fetch operation.
569:     ///
570:     /// Verifies the proof against the current root first, then falls back
571:     /// to a matching historical root from `retained_roots` if available.
572:     fn handle_fetch_result(
573:         &mut self,
574:         fetch_result: IndexedFetchResult<DB::Family, DB::Op, DB::Digest, S::Error>,
575:     ) -> Result<(), Error<DB, S>> {
576:         // Removal aborts a request's future, so a result for an untracked ID should
577:         // be unreachable.
578:         let Some(request) = self.outstanding_requests.remove(fetch_result.id) else {
579:             return Ok(());
580:         };
581: 
582:         let (response, feedback_tx) = fetch_result.result.map_err(SyncError::Source)?;
583: 
584:         let start_loc = request.start();
585:         let size = request.size();
586: 
587:         // The proof must cover exactly the requested size.
588:         if response.proof().leaves != size {
589:             return Self::reject_response(feedback_tx);
590:         }
591:         // A response must match the shape of its request.
592:         match (request, response) {
593:             (Request::Operations { max_ops, .. }, Response::Operations { proof, operations }) => {
594:                 let operations_len = operations.len() as u64;
595:                 if operations_len == 0 || operations_len > max_ops.get() {
596:                     return Self::reject_response(feedback_tx);
597:                 }
598:                 let Some(root) = self.verification_root(size) else {
599:                     return Ok(());
600:                 };
601:                 let elements = operations.iter().map(|op| op.encode()).collect::<Vec<_>>();
602:                 if !proof.verify_range_inclusion(&self.hasher, &elements, start_loc, root) {
603:                     return Self::reject_response(feedback_tx);
604:                 }
605:                 if let Some(feedback_tx) = feedback_tx {
606:                     feedback_tx.send_lossy(true);
607:                 }
608:                 self.store_operations(start_loc, operations);
609:             }
610:             (
611:                 Request::Boundary { .. },
612:                 Response::Boundary {
613:                     proof,
614:                     op,
615:                     pinned_nodes,
616:                 },
617:             ) => {
618:                 // Use the pinned nodes only if the current target still needs them. Otherwise
619:                 // keep the operation and drop the pinned nodes.
620:                 let need_pinned = size == self.target.range.end()
621:                     && self.pinned_nodes.is_none()
622:                     && start_loc == self.target.range.start();
623:                 let element = [op.encode()];
624:                 let valid = if need_pinned {
625:                     proof.verify_proof_and_pinned_nodes(
626:                         &self.hasher,
627:                         &element,
628:                         start_loc,
629:                         &pinned_nodes,
630:                         &self.target.root,
631:                     )
632:                 } else {
633:                     let Some(root) = self.verification_root(size) else {
634:                         return Ok(());
635:                     };
636:                     proof.verify_range_inclusion(&self.hasher, &element, start_loc, root)
637:                 };
638:                 if !valid {
639:                     if need_pinned {
640:                         tracing::warn!("boundary response failed verification");
641:                     }
642:                     return Self::reject_response(feedback_tx);
643:                 }
644:                 if let Some(feedback_tx) = feedback_tx {
645:                     feedback_tx.send_lossy(true);
646:                 }
647:                 if need_pinned {
648:                     self.pinned_nodes = Some(pinned_nodes);
649:                 }
650:                 self.store_operations(start_loc, vec![op]);
651:             }
652:             _ => return Self::reject_response(feedback_tx),
653:         }
654: 
655:         Ok(())
656:     }
657: 
658:     /// The root to verify a response against at a given size.
659:     fn verification_root(&self, size: Location<DB::Family>) -> Option<&DB::Digest> {
660:         if size == self.target.range.end() {
661:             Some(&self.target.root)
662:         } else {
663:             self.retained_roots.get(&size)
664:         }
665:     }

667:     /// Handle a sync event and return the next engine state.
668:     async fn handle_event(
669:         mut self,
670:         event: Event<DB::Family, DB::Op, DB::Digest, S::Error>,
671:     ) -> Result<NextStep<Self, DB>, Error<DB, S>> {
672:         match event {
673:             Event::TargetUpdate(new_target) => {
674:                 // A non-advancing update is discarded.
675:                 if !new_target.advances(&self.target) {
676:                     return Ok(NextStep::Continue(self));
677:                 }
678:                 // A same-root update that advances is impossible for an append-only log and
679:                 // indicates a caller bug.
680:                 if new_target.root == self.target.root {
681:                     return Err(SyncError::Engine(EngineError::SyncTargetRootUnchanged));
682:                 }
683: 
684:                 let mut updated_self = self.reset_for_target_update(new_target).await?;
685:                 updated_self.record_progress();
686:                 updated_self.schedule_requests()?;
687:                 Ok(NextStep::Continue(updated_self))
688:             }
689:             Event::UpdateChannelClosed => {
690:                 self.update_rx = None;
691:                 Ok(NextStep::Continue(self))
692:             }
693:             Event::FinishRequested => {
694:                 self.finish_rx = None;
695:                 self.finish_requested = true;
696:                 Ok(NextStep::Continue(self))
697:             }
698:             Event::FinishChannelClosed => Err(SyncError::Engine(EngineError::FinishChannelClosed)),
699:             Event::BatchReceived(fetch_result) => {
700:                 // An aborted request carries no result, but still wakes the loop to reschedule.
701:                 if let Ok(fetch_result) = fetch_result {
702:                     self.handle_fetch_result(fetch_result)?;
703:                 }
704:                 self.schedule_requests()?;
705:                 let mut engine = self.apply_operations().await?;
706:                 engine.record_progress();
707:                 Ok(NextStep::Continue(engine))
708:             }
709:         }
710:     }

712:     /// Execute one step of the synchronization process.
713:     ///
714:     /// This is the main coordination method that:
715:     /// 1. Checks if sync is complete
716:     /// 2. Waits for the next synchronization event
717:     /// 3. Handles different event types (target updates, fetch results)
718:     /// 4. Coordinates request scheduling and operation application
719:     ///
720:     /// Returns `NextStep::Complete(database)` when sync is finished, or
721:     /// `NextStep::Continue(self)` when more work remains.
722:     #[boxed]
723:     pub(crate) async fn step(mut self) -> Result<NextStep<Self, DB>, Error<DB, S>> {
724:         self.drain_finish_requests()?;
725: 
726:         // Check if sync is complete
727:         if self.is_ready_to_complete()? {
728:             // Take a queued target update before completing at the old target, unless the
729:             // caller already asked to finish. Updates that do not advance the target are
730:             // discarded.
731:             if !self.finish_requested {
732:                 while let Some(update_rx) = self.update_rx.as_mut() {
733:                     match update_rx.try_recv() {
734:                         Ok(new_target) => {
735:                             if new_target.advances(&self.target) {
736:                                 return self.handle_event(Event::TargetUpdate(new_target)).await;
737:                             }
738:                         }
739:                         Err(TryRecvError::Empty) => break,
740:                         Err(TryRecvError::Disconnected) => {
741:                             self.update_rx = None;
742:                         }
743:                     }
744:                 }
745:             }
746: 
747:             self.report_reached_target().await;
748: 
749:             if self.finish_rx.is_some() {
750:                 let event = wait_for_event(
751:                     &mut self.update_rx,
752:                     &mut self.finish_rx,
753:                     &mut self.outstanding_requests,
754:                 )
755:                 .await
756:                 .ok_or(SyncError::Engine(EngineError::SyncStalled))?;
757:                 return self.handle_event(event).await;
758:             }
759: 
760:             return Ok(NextStep::Complete(self.complete().await?));
761:         }
762: 
763:         // Wait for the next synchronization event
764:         let event = wait_for_event(
765:             &mut self.update_rx,
766:             &mut self.finish_rx,
767:             &mut self.outstanding_requests,
768:         )
769:         .await
770:         .ok_or(SyncError::Engine(EngineError::SyncStalled))?;
771:         self.handle_event(event).await
772:     }
773: 
774:     /// Build the final database from the completed sync and verify its root against the
775:     /// target.
776:     async fn complete(mut self) -> Result<DB, Error<DB, S>> {
777:         self.journal = self.journal.sync().await?;
778: 
779:         let database = DB::from_sync_result(
780:             self.context,
781:             self.config,
782:             self.journal,
783:             self.pinned_nodes,
784:             self.target.range.clone(),
785:             self.apply_batch_size,
786:         )
787:         .await?;
788: 
789:         let got_root = database.root();
790:         let expected_root = self.target.root;
791:         if got_root != expected_root {
792:             return Err(SyncError::Engine(EngineError::RootMismatch {
793:                 expected: expected_root,
794:                 actual: got_root,
795:             }));
796:         }
797: 
798:         Ok(database.persist_sync_result().await?)
799:     }
800: 
801:     /// Run sync to completion, returning the final database when done.
802:     ///
803:     /// This method repeatedly calls `step()` until sync is complete. The `step()` method
804:     /// handles building the final database and verifying the root digest.
805:     pub async fn sync(mut self) -> Result<DB, Error<DB, S>> {
806:         // Run sync loop until completion
807:         loop {
808:             match self.step().await? {
809:                 NextStep::Continue(new_engine) => self = new_engine,
810:                 NextStep::Complete(database) => return Ok(database),
811:             }
812:         }
813:     }
814: }
