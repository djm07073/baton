```rust
0: //! Traits for database batch lifecycle and state sync in [`Stateful`](super::Stateful).
1: //!
2: //! This module defines the boundary between stateful application logic and
3: //! storage backends (QMDB variants).
4: //!
5: //! # Batch Lifecycle
6: //!
7: //! Normal execution has three stages:
8: //! 1. [`Unmerkleized`]: mutable, in-progress batch (concrete types expose reads and writes).
9: //! 2. [`Merkleized`]: a sealed batch with a computed root.
10: //! 3. Finalization applies the sealed batch via [`ManagedDb::apply`]. It then requests durability
11: //!    via [`ManagedDb::finalize`] and observes completion through [`Barrier`]. A barrier covers the
12: //!    state applied before it was requested. Later batches may be applied while it is pending.
13: //!
14: //! [`DatabaseSet`] groups one or more [`ManagedDb`] instances into one logical
15: //! unit for execution and commit.
16: //!
17: //! # State Sync
18: //!
19: //! State sync orchestration is expressed by two traits:
20: //! - [`StateSyncDb`]: per-database sync entrypoint.
21: //! - [`StateSyncSet`]: set-level orchestration.
22: //!
23: //! ## Anchors
24: //!
25: //! Each set of sync targets is paired with an anchor `(Height, Round, D)` where
26: //! `D` is the block digest. The db layer never interprets the anchor; it
27: //! only tracks which anchor each database converged on.
28: //!
29: //! On completion, [`StateSyncSet::sync`] returns the anchor that all databases
30: //! agreed on. The caller uses this to set the marshal floor and the
31: //! last-processed digest, ensuring they match the actual convergence point
32: //! rather than whatever marshal's head happens to be (which may have advanced
33: //! during sync).
34: //!
35: //! ## Convergence Algorithm (tuple sets)
36: //!
37: //! Tuple [`StateSyncSet`] implementations assign each `(anchor, targets)`
38: //! pair a *generation* number and use this algorithm:
39: //!
40: //! 1. Forward tip updates only to databases that have not yet reported
41: //!    "reached target". Reached databases are frozen to prevent them from
42: //!    running ahead to a newer anchor.
43: //! 2. When all databases report reached, compare the generation each was
44: //!    assigned when it reported.
45: //! 3. If all generations match, every database synced to targets from the
46: //!    same anchor. Return that anchor.
47: //! 4. If generations differ, *regroup*: re-send the highest-reached
48: //!    generation's targets to the behind databases, clear their reached
49: //!    state, and repeat from step 1.
50: //!
51: //! ### Chasing a moving tip
52: //!
53: //! ```text
54: //! time -------------------------------------------------------------->
55: //!
56: //! marshal finalized tip:   A0 ------ A1 ------ A2 ------ A3
57: //! generation:              g0        g1        g2        g3
58: //!
59: //! db0 (slow):              g0 ------------------> g1 -----------------> g3 reached
60: //! db1 (fast):              g0 ----> g1 reached -- frozen -- regroup --> g3 reached
61: //! db2 (fast):              g0 ----> g1 reached -- frozen -- regroup --> g3 reached
62: //!
63: //! coordinator queue while db0 is still catching up:
64: //!                          [A2] [A3] -- drain --> keep only A3
65: //!
66: //! finish only when:
67: //! - every database has reported the same generation
68: //! - no newer tip update is still queued behind it
69: //! ```
70: //!
71: //! The coordinator continuously drains tip updates and keeps only the latest
72: //! value before forwarding, which avoids target-channel backpressure buildup.
73: //! The `generation_state` map is pruned after every dispatch to only retain
74: //! generations currently assigned to at least one database, so memory usage
75: //! is bounded by the number of databases regardless of how long sync runs.
76:

112: /// A database shared across tasks.
113: ///
114: /// Owned mutations (apply, finalize, prune, rewind) take the database out of
115: /// the cell under the write lock ([Self::write]) and put it back on success
116: /// ([WriteSlot::put]). A failure, panic, or cancellation mid-operation leaves
117: /// the cell empty permanently, and every later [Self::read] or [Self::write]
118: /// panics. A lost database is fatal here by design and requires a restart.

327: /// One database managed by the [`Stateful`](super::Stateful) wrapper.
328: ///
329: /// Implementations create new batches from applied state, apply finalized
330: /// batches back to storage, and start durability work separately.
331: ///
332: /// [`new_batch`](Self::new_batch) consumes origin-bound read access so batch
333: /// types can snapshot applied state and retain the matching [`Shared`] handle.
334: ///
335: /// `E` is a trait generic (not an associated type), so one database type can
336: /// work across runtimes that satisfy the bounds.
337: ///
338: /// # Ownership
339: ///
340: /// Mutating methods take the database by value and return it on success. If a mutating
341: /// method returns an error, or its future is dropped before it finishes, the database is
342: /// gone: state that was not yet durable is discarded, but everything already on disk stays
343: /// recoverable.
344: pub trait ManagedDb<E>: Send + Sync + Sized {
345:     /// An in-progress batch of mutations that has not yet been merkleized.
346:     type Unmerkleized: Unmerkleized;
347: 
348:     /// A batch whose root has been computed but has not yet been applied to
349:     /// the underlying database.
350:     ///
351:     /// Constrained so that [`Merkleized::new_batch`] produces the same
352:     /// [`Unmerkleized`] type as [`ManagedDb::new_batch`](Self::new_batch).
353:     /// Cloning must preserve the same sealed branch state and should be cheap.
354:     type Merkleized: Clone + Merkleized<Unmerkleized = Self::Unmerkleized>;
355: 
356:     /// The error type returned by fallible operations.
357:     type Error: Debug + Send;
358: 
359:     /// Configuration needed to construct a new database instance.
360:     type Config: Send;
361: 
362:     /// Sync target type for state sync of this database.
363:     ///
364:     /// Typically a database-specific state commitment plus the operation range needed to reach it.
365:     type SyncTarget: Clone + PartialEq + Send + Sync;
366: 
367:     /// Construct a new database from its configuration.
368:     fn init(
369:         context: E,
370:         config: Self::Config,
371:     ) -> impl Future<Output = Result<Self, Self::Error>> + Send;
372: 
373:     /// Return the sync target produced by a newly initialized database.
374:     ///
375:     /// This must match [`sync_target`](Self::sync_target) after opening an empty partition.
376:     fn initial_sync_target() -> Self::SyncTarget;
377: 
378:     /// Create a new unmerkleized batch rooted at the read-locked database's
379:     /// applied state.
380:     ///
381:     /// This method must return without retaining `database`, releasing its read
382:     /// lock before the batch performs any lazy read-through work. Batch types
383:     /// can retain the matching handle returned by [`BatchContext::into_parts`].
384:     fn new_batch(database: BatchContext<'_, Self>) -> Self::Unmerkleized;
385: 
386:     /// Return true if a merkleized batch matches a sync target.
387:     fn matches_sync_target(batch: &Self::Merkleized, target: &Self::SyncTarget) -> bool;
388: 
389:     /// Apply a merkleized batch's changeset to the underlying database.
390:     ///
391:     /// In QMDB, this encapsulates calling `merkleized.finalize()` to produce
392:     /// a `Changeset`, then `db.apply_batch(changeset)`. The returned database
393:     /// must expose the batch as an independently rewindable recovery checkpoint.
394:     /// The checkpoint need not be durable until the handle returned by
395:     /// [`Self::finalize`] resolves.
396:     fn apply(
397:         self,
398:         batch: Self::Merkleized,
399:     ) -> impl Future<Output = Result<Self, Self::Error>> + Send;
400: 
401:     /// Begin persisting every checkpoint applied before this call.
402:     ///
403:     /// Awaiting the returned handle proves the state applied before this call
404:     /// durable. Later batches may be applied while the handle is pending, but
405:     /// that dirty suffix needs a subsequent finalization. The caller must
406:     /// observe the handle before finalizing again.
407:     fn finalize(self) -> impl Future<Output = Result<(Self, Handle<()>), Self::Error>> + Send;
408: 
409:     /// Prune the database to a previously finalized sync target.
410:     ///
411:     /// The caller resolves every handle returned by [`Self::finalize`] before pruning. Databases
412:     /// that do not retain pruneable operation history can rely on the default no-op. Any pruning
413:     /// effects must be durable before returning.
414:     fn prune(
415:         self,
416:         _target: &Self::SyncTarget,
417:     ) -> impl Future<Output = Result<Self, Self::Error>> + Send {
418:         async { Ok(self) }
419:     }
420: 
421:     /// Return the latest applied recovery target.
422:     ///
423:     /// Returning a target does not by itself prove that target durable.
424:     fn sync_target(&self) -> Self::SyncTarget;
425: 
426:     /// Rewind applied state to `target`.
427:     ///
428:     /// Implementations must ensure rewind effects are durable before returning
429:     /// the database (for example by committing after rewind).
430:     fn rewind_to_target(
431:         self,
432:         target: Self::SyncTarget,
433:     ) -> impl Future<Output = Result<Self, Self::Error>> + Send;
434: }

436: /// Durability barrier returned by [`DatabaseSet::finalize`].
437: ///
438: /// Holds one [`ManagedDb::finalize`] handle per database in the set. Deferred
439: /// sync failures surface only here, so every barrier must be awaited via
440: /// [`durable`](Self::durable), typically on a futures pool. Before
441: /// [`DatabaseSet::prune`] runs, the active barrier must resolve.
442: ///
443: /// # Examples
444: ///
445: /// ```
446: /// use commonware_glue::stateful::db::Barrier;
447: /// use commonware_runtime::Handle;
448: ///
449: /// struct CustomDatabaseSet;
450: ///
451: /// # async fn example() {
452: /// let barrier = Barrier::from_handles::<CustomDatabaseSet>([
453: ///     Handle::ready(Ok(())),
454: /// ]);
455: /// assert!(barrier.durable().await);
456: /// # }
457: /// ```
458: #[must_use = "await `durable` to surface deferred sync failures"]
459: pub struct Barrier {
460:     syncs: Vec<(&'static str, Option<usize>, Handle<()>)>,
461: }
462: 
463: impl Barrier {
464:     /// Construct a barrier from deferred sync handles owned by `T`.
465:     ///
466:     /// Failures identify `T` and the handle's zero-based position in the
467:     /// provided iteration. An empty barrier is immediately durable.
468:     pub fn from_handles<T: ?Sized>(handles: impl IntoIterator<Item = Handle<()>>) -> Self {
469:         let db_type = std::any::type_name::<T>();
470:         Self {
471:             syncs: handles
472:                 .into_iter()
473:                 .enumerate()
474:                 .map(|(index, handle)| (db_type, Some(index), handle))
475:                 .collect(),
476:         }
477:     }
478: 
479:     /// Resolves `true` once every deferred sync is durable.
480:     ///
481:     /// A sync failure panics because the database has already advanced past
482:     /// non-durable state. Returns `false` only when runtime shutdown aborts
483:     /// or closes a sync handle.
484:     pub async fn durable(self) -> bool {
485:         let syncs = self
486:             .syncs
487:             .into_iter()
488:             .map(|(db_type, index, handle)| async move {
489:                 match handle.await {
490:                     Ok(()) => Ok(true),
491:                     Err(RuntimeError::Closed | RuntimeError::Aborted) => {
492:                         debug!(db_type, "runtime shutdown before database sync completed");
493:                         Ok(false)
494:                     }
495:                     Err(err) => Err((db_type, index, err)),
496:                 }
497:             });
498: 
499:         match try_join_all(syncs).await {
500:             Ok(results) => results.into_iter().all(|durable| durable),
501:             Err((db_type, index, err)) => {
502:                 let index = index.map_or(String::new(), |i| format!("index {i}, "));
503:                 panic!("database sync failed ({index}type {db_type}): {err}");
504:             }
505:         }
506:     }
507: }

509: /// A collection of individually locked [`ManagedDb`] instances.
510: ///
511: /// Each database is wrapped in [`Shared`], so the set is cheap to
512: /// clone and each database can be shared without a global lock.
513: /// Multi-database mutations must not hold one member's writer while waiting
514: /// to acquire another. Readers may span members in the opposite order.
515: ///
516: /// `E` is a trait generic (not an associated type), so one set type can work
517: /// across runtimes that satisfy the bounds.
518: ///
519: /// # Mutation Safety
520: ///
521: /// Calls to [`Self::apply`], [`Self::finalize`], [`Self::prune`], and
522: /// [`Self::rewind_to_targets`] must not overlap. Implementations panic if an
523: /// underlying database mutation returns an error. If a mutation panics or is
524: /// cancelled after taking a database from [`Shared`], the affected cell remains
525: /// empty and subsequent access panics.
526: pub trait DatabaseSet<E>: Clone + Send + Sync + 'static {
527:     /// Tuple of [`ManagedDb::Unmerkleized`] for every database in the set.
528:     type Unmerkleized: Send;
529: 
530:     /// Tuple of [`ManagedDb::Merkleized`] for every database in the set.
531:     /// Cloning must preserve the same sealed branch state and should be cheap.
532:     type Merkleized: Clone + Send + Sync;
533: 
534:     /// Read-only handles for observing the applied database state.
535:     ///
536:     /// Implementations must not expose mutation capabilities through this
537:     /// type. In particular, readers must not construct or apply batches,
538:     /// finalize database state, prune, or rewind.
539:     type Readers: Send;
540: 
541:     /// Configuration needed to construct every database in the set.
542:     ///
543:     /// - Single database sets use that database's [`ManagedDb::Config`].
544:     /// - Multi-database tuple sets use a tuple of per-database configs
545:     ///   `(Db1::Config, Db2::Config, ...)`.
546:     type Config: Send;
547: 
548:     /// Per-database sync targets extracted from a finalized block.
549:     ///
550:     /// For a single-database set this is one target. For multi-database sets it is a tuple of
551:     /// targets, one per database.
552:     type SyncTargets: Clone + PartialEq + Send + Sync;
553: 
554:     /// Construct the database set from its configuration.
555:     fn init(context: E, config: Self::Config) -> impl Future<Output = Self> + Send;
556: 
557:     /// Return the sync targets produced by a newly initialized database set.
558:     fn initial_sync_targets() -> Self::SyncTargets;
559: 
560:     /// Create unmerkleized batches from each database's applied state.
561:     ///
562:     /// Implementations must release every read lock before the returned
563:     /// batches perform lazy reads.
564:     fn new_batches(&self) -> impl Future<Output = Self::Unmerkleized> + Send;
565: 
566:     /// Create child unmerkleized batches from a pending merkleized parent.
567:     ///
568:     /// No lock is needed; reads come from the in-memory merkleized state.
569:     fn fork_batches(parent: &Self::Merkleized) -> Self::Unmerkleized;
570: 
571:     /// Return true if merkleized batches match the sync targets.
572:     fn matches_sync_targets(batches: &Self::Merkleized, targets: &Self::SyncTargets) -> bool;
573: 
574:     /// Return read-only handles for every database in the set.
575:     fn readers(&self) -> Self::Readers;
576: 
577:     /// Apply each merkleized batch's changeset.
578:     ///
579:     /// Returns once every database exposes its batch as an independently
580:     /// rewindable recovery checkpoint. Durability is started separately with
581:     /// [`DatabaseSet::finalize`].
582:     fn apply(&self, batches: Self::Merkleized) -> impl Future<Output = ()> + Send;
583: 
584:     /// Begin persisting every checkpoint applied before this call.
585:     ///
586:     /// The returned [`Barrier`] resolves once the captured state is durable in
587:     /// every database. Later batches may be applied while the barrier is
588:     /// pending, but that dirty suffix needs a subsequent finalization. The
589:     /// barrier must be observed before finalizing again.
590:     fn finalize(&self) -> impl Future<Output = Barrier> + Send;
591: 
592:     /// Prune each database to the provided per-database targets.

804: impl<E, T, R, D> StateSyncSet<E, R, D> for Shared<T>
805: where
806:     E: Send + Sync + Metrics,
807:     T: StateSyncDb<E, R> + 'static,
808:     R: Send + 'static,
809:     D: Digest,
810: {
811:     type Error = T::SyncError;
812: 
813:     #[allow(clippy::too_many_arguments)]
814:     async fn sync(
815:         context: E,
816:         config: Self::Config,
817:         source: R,
818:         anchor: Anchor<D>,
819:         target: Self::SyncTargets,
820:         tip_updates: ring::Receiver<TipUpdate<D, Self::SyncTargets>>,
821:         sync_config: SyncEngineConfig,
822:     ) -> Result<(Self, Anchor<D>), Self::Error> {
823:         let (target_tx, target_rx) = mpsc::channel(sync_config.update_channel_size.get());
824:         let (finish_tx, finish_rx) = mpsc::channel(1);
825:         let (reached_tx, mut reached_rx) = mpsc::channel(1);
826:         let mut current_target = target.clone();
827:         let sync = T::sync_db(
828:             context,
829:             config,
830:             source,
831:             target,
832:             target_rx,
833:             Some(finish_rx),
834:             Some(reached_tx),
835:             sync_config,
836:         );
837: 
838:         let coordinator = async {
839:             let mut current_anchor = anchor;
840:             let mut tip_updates = Some(tip_updates);
841:             loop {
842:                 if !drain_single_tip_updates(
843:                     &mut tip_updates,
844:                     &target_tx,
845:                     &mut current_anchor,
846:                     &mut current_target,
847:                 )
848:                 .await
849:                 {
850:                     return (current_anchor, current_target);
851:                 }
852: 
853:                 let update_future = tip_updates.as_mut().map_or_else(
854:                     || Either::Right(pending()),
855:                     |updates| Either::Left(updates.recv()),
856:                 );
857:                 select! {
858:                     reached = reached_rx.recv() => {
859:                         let Some(reached) = reached else {
860:                             return (current_anchor, current_target);
861:                         };
862:                         if !drain_single_tip_updates(
863:                             &mut tip_updates,
864:                             &target_tx,
865:                             &mut current_anchor,
866:                             &mut current_target,
867:                         )
868:                         .await
869:                         {
870:                             return (current_anchor, current_target);
871:                         };
872:                         if reached != current_target {
873:                             continue;
874:                         }
875:                         let _ = finish_tx.send_lossy(()).await;
876:                         return (current_anchor, current_target);
877:                     },
878:                     update = update_future => {
879:                         let Some(update) = update else {
880:                             tip_updates = None;
881:                             continue;
882:                         };
883:                         let target = update.record(|new_anchor, new_target| {
884:                             if new_anchor.height <= current_anchor.height {
885:                                 return None;
886:                             }
887:                             current_anchor = new_anchor;
888:                             if new_target == current_target {
889:                                 return None;
890:                             }
891:                             current_target = new_target.clone();
892:                             Some(new_target)
893:                         });
894:                         let Some(new_target) = target else {
895:                             continue;
896:                         };
897:                         if !target_tx.send_lossy(new_target).await {
898:                             return (current_anchor, current_target);
899:                         }
900:                     },
901:                 }
902:             }
903:         };
904: 
905:         let (db_result, (converged_anchor, converged_target)) = join!(sync, coordinator);
906:         let database = db_result?;
907:         assert!(
908:             T::sync_target(&database) == converged_target,
909:             "state sync database target does not match the coordinator target",
910:         );
911:         Ok((Self::new("stateful.db", database), converged_anchor))
912:     }
913: }

915: async fn drain_single_tip_updates<D, T>(
916:     tip_updates: &mut Option<ring::Receiver<TipUpdate<D, T>>>,
917:     target_tx: &mpsc::Sender<T>,
918:     current_anchor: &mut Anchor<D>,
919:     current_target: &mut T,
920: ) -> bool
921: where
922:     D: Digest,
923:     T: Clone + PartialEq + Send + Sync,
924: {
925:     let mut drained = 0usize;
926:     let mut latest = None;
927:     loop {
928:         let update = match tip_updates.as_mut().map(ring::Receiver::try_recv) {
929:             Some(Ok(update)) => update,
930:             Some(Err(ring::TryRecvError::Empty)) => break,
931:             Some(Err(ring::TryRecvError::Disconnected)) => {
932:                 *tip_updates = None;
933:                 break;
934:             }
935:             None => break,
936:         };
937:         drained += 1;
938: 
939:         update.record(|new_anchor, new_target| {
940:             let latest_height = latest
941:                 .as_ref()
942:                 .map_or(current_anchor.height, |(anchor, _): &(Anchor<D>, T)| {
943:                     anchor.height
944:                 });
945:             if new_anchor.height > latest_height {
946:                 latest = Some((new_anchor, new_target));
947:             }
948:         });
949:         if drained.is_multiple_of(MAX_CHANNEL_DRAIN_PER_TICK) {
950:             reschedule().await;
951:         }
952:     }
953: 
954:     let Some((new_anchor, new_target)) = latest else {
955:         return true;
956:     };
957:     *current_anchor = new_anchor;
958:     if new_target == *current_target {
959:         return true;
960:     }
961:     *current_target = new_target.clone();
962:     target_tx.send_lossy(new_target).await
963: }

1459: async fn drain_generation_updates<T>(
1460:     generation_rx: &mut Option<mpsc::Receiver<(usize, T)>>,
1461:     current_generation: &mut usize,
1462:     current_target: &mut T,
1463:     last_reached_target: &Option<T>,
1464:     last_reported_generation: &mut Option<usize>,
1465:     reached_event_sender: &mpsc::Sender<(usize, usize)>,
1466:     idx: usize,
1467: ) where
1468:     T: Clone + PartialEq,
1469: {
1470:     if let Some(updates) = generation_rx.as_mut() {
1471:         let mut drained = 0usize;
1472:         loop {
1473:             match updates.try_recv() {
1474:                 Ok((generation, target)) => {
1475:                     drained += 1;
1476:                     *current_generation = generation;
1477:                     *current_target = target;
1478: 
1479:                     if last_reached_target.as_ref() == Some(current_target)
1480:                         && *last_reported_generation != Some(*current_generation)
1481:                     {
1482:                         if !reached_event_sender
1483:                             .send_lossy((idx, *current_generation))
1484:                             .await
1485:                         {
1486:                             return;
1487:                         }
1488:                         *last_reported_generation = Some(*current_generation);
1489:                     }
1490:                     if drained.is_multiple_of(MAX_CHANNEL_DRAIN_PER_TICK) {
1491:                         reschedule().await;
1492:                     }
1493:                 }
1494:                 Err(mpsc::error::TryRecvError::Empty) => break,
1495:                 Err(mpsc::error::TryRecvError::Disconnected) => {
1496:                     *generation_rx = None;
1497:                     break;
1498:                 }
1499:             }
1500:         }
1501:     }
1502: }

1504: /// Per-database sync tracking state.
1505: #[derive(Clone, Copy, Debug, PartialEq, Eq)]
1506: enum DbSyncState {

1509:     /// Database reported it reached its assigned generation's targets.
1510:     Reached { generation: usize },
1511: }
1512: 
1513: impl DbSyncState {
1514:     const fn generation(self) -> usize {
1515:         match self {
1516:             Self::Seeking { generation } | Self::Reached { generation } => generation,
1517:         }
1518:     }
1519: 
1520:     const fn is_reached(self) -> bool {
1521:         matches!(self, Self::Reached { .. })
1522:     }
1523: }

1525: /// What the coordinator should do after processing events.
1526: enum CoordinatorAction<D: Digest, T> {
1527:     /// Nothing to do; wait for the next event.
1528:     Wait,
1529:     /// Dispatch targets to non-reached databases for `generation`.
1530:     Dispatch { generation: usize, targets: T },
1531:     /// All databases converged on the same generation.
1532:     Converged { anchor: Anchor<D>, targets: T },
1533: }
1534: 
1535: /// Pure state machine for multi-database sync convergence.
1536: ///
1537: /// Tracks which generation each database is assigned to, which have
1538: /// reported "reached", and decides when to regroup or declare
1539: /// convergence.
1540: struct CoordinatorState<D: Digest, T> {
1541:     dbs: Vec<DbSyncState>,
1542:     generation_state: BTreeMap<usize, (Anchor<D>, T)>,
1543:     current_generation: usize,
1544:     latest_tip: Option<(Anchor<D>, T)>,
1545:     last_dispatched_anchor: Anchor<D>,
1546: }
1547: 
1548: impl<D: Digest, T: Clone> CoordinatorState<D, T> {
1549:     fn new(db_count: usize, anchor: Anchor<D>, targets: T) -> Self {
1550:         let dbs = vec![DbSyncState::Seeking { generation: 0 }; db_count];
1551:         let mut generation_state = BTreeMap::new();
1552:         generation_state.insert(0, (anchor, targets));
1553:         Self {
1554:             dbs,
1555:             generation_state,
1556:             current_generation: 0,
1557:             latest_tip: None,
1558:             last_dispatched_anchor: anchor,
1559:         }
1560:     }
1561: 
1562:     /// Record that database `idx` reached `generation`.
1563:     ///
1564:     /// Reached events can arrive late. If the database has already been
1565:     /// re-assigned to a newer generation, stale events are ignored.
1566:     fn record_reached(&mut self, idx: usize, generation: usize) {
1567:         if self.dbs[idx].generation() != generation {
1568:             return;
1569:         }
1570:         if self.dbs[idx].is_reached() {
1571:             return;
1572:         }
1573:         self.dbs[idx] = DbSyncState::Reached { generation };
1574:     }
1575: 
1576:     /// Record a new tip update.
1577:     ///
1578:     /// Sync targets must move strictly forward. Ignore stale and duplicate
1579:     /// anchors to avoid dispatching backward targets.
1580:     fn record_tip_update(&mut self, anchor: Anchor<D>, targets: T) {
1581:         let current_height = self
1582:             .latest_tip
1583:             .as_ref()
1584:             .map_or(self.last_dispatched_anchor.height, |(latest_anchor, _)| {
1585:                 latest_anchor.height
1586:             });
1587:         if anchor.height <= current_height {
1588:             return;
1589:         }
1590:         self.latest_tip = Some((anchor, targets));
1591:     }
1592: 
1593:     /// Determine the next action. Mutates internal state for regroup/dispatch.
1594:     ///
1595:     /// Returns which database indices should receive targets via
1596:     /// `dbs[idx].is_reached() == false` after a `Dispatch` action.
1597:     fn next_action(&mut self) -> CoordinatorAction<D, T> {
1598:         let all_reached = self.dbs.iter().all(|db| db.is_reached());
1599: 
1600:         if all_reached {
1601:             let min_gen = self.dbs.iter().map(|db| db.generation()).min().unwrap();
1602:             let max_gen = self.dbs.iter().map(|db| db.generation()).max().unwrap();
1603: 
1604:             if min_gen == max_gen {
1605:                 if let Some((anchor, targets)) = self.latest_tip.take() {
1606:                     let generation = self.current_generation + 1;
1607:                     self.current_generation = generation;
1608:                     self.dbs.fill(DbSyncState::Seeking { generation });
1609:                     self.generation_state
1610:                         .insert(generation, (anchor, targets.clone()));
1611:                     self.last_dispatched_anchor = anchor;
1612:                     self.prune_generations();
1613:                     return CoordinatorAction::Dispatch {
1614:                         generation,
1615:                         targets,
1616:                     };
1617:                 }
1618: 
1619:                 let (anchor, targets) = self
1620:                     .generation_state
1621:                     .get(&min_gen)
1622:                     .expect("missing state for converged generation")
1623:                     .clone();
1624:                 return CoordinatorAction::Converged { anchor, targets };
1625:             }
1626: 
1627:             // Regroup: reset behind databases to seek the highest generation.
1628:             let (_anchor, targets) = self
1629:                 .generation_state
1630:                 .get(&max_gen)
1631:                 .expect("missing state for regroup generation")
1632:                 .clone();
1633:             for db in &mut self.dbs {
1634:                 if db.generation() != max_gen {
1635:                     *db = DbSyncState::Seeking {
1636:                         generation: max_gen,
1637:                     };
1638:                 }
1639:             }
1640:             self.prune_generations();
1641:             return CoordinatorAction::Dispatch {
1642:                 generation: max_gen,
1643:                 targets,
1644:             };
1645:         }
1646: 
1647:         // Not all reached. If there's a pending tip, dispatch it.
1648:         let Some((anchor, targets)) = self.latest_tip.take() else {
1649:             return CoordinatorAction::Wait;
1650:         };
1651: 
1652:         let generation = self.current_generation + 1;
1653:         self.current_generation = generation;
1654:         for db in &mut self.dbs {
1655:             if !db.is_reached() {
1656:                 *db = DbSyncState::Seeking { generation };
1657:             }
1658:         }
1659:         self.generation_state
1660:             .insert(generation, (anchor, targets.clone()));
1661:         self.last_dispatched_anchor = anchor;
1662: 
1663:         self.prune_generations();
1664:         CoordinatorAction::Dispatch {
1665:             generation,
1666:             targets,
1667:         }
1668:     }
1669: 
1670:     /// Retain only generations referenced by at least one database.
1671:     fn prune_generations(&mut self) {
1672:         self.generation_state
1673:             .retain(|r#gen, _| self.dbs.iter().any(|db| db.generation() == *r#gen));
1674:     }
1675: 
1676:     /// Whether database `idx` is a non-reached recipient for dispatch.
1677:     fn should_dispatch(&self, idx: usize) -> bool {
1678:         !self.dbs[idx].is_reached()
1679:     }
1680: 
1681:     /// Advance a reached database to `generation` when its target is unchanged.
1682:     fn mark_reached_same_target(&mut self, idx: usize, generation: usize) {
1683:         if !self.dbs[idx].is_reached() {
1684:             return;
1685:         }
1686:         self.dbs[idx] = DbSyncState::Reached { generation };
1687:     }
1688: }

1773:     }));

1777:         context.child("compact_reached").spawn(move |_| async move {
1778:             while let Some(reached_engine_target) = rx.recv().await {

1784:                     break;

1810: async fn apply<E, T: ManagedDb<E>>(database: T, batch: T::Merkleized, index: Option<usize>) -> T {
1811:     // Mutable apply failures are fatal because the batch may already have been
1812:     // applied to other databases in the same set, leaving partially applied state.
1813:     match database.apply(batch).await {
1814:         Ok(result) => result,
1815:         Err(err) => {
1816:             let index = index.map_or(String::new(), |i| format!("index {i}, "));
1817:             panic!(
1818:                 "database apply failed ({index}type {}): {err:?}",
1819:                 core::any::type_name::<T>(),
1820:             );
1821:         }
1822:     }
1823: }
