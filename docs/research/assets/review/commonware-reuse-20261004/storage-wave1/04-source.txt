247:     /// Expand this staged batch with more reads.
248:     ///
249:     /// Existing read indices remain stable. Newly read keys are appended to the staged read set and
250:     /// assigned the returned range. Expansion does not deduplicate against previously staged keys
251:     /// and does not observe values computed for earlier staged slots but not yet passed to
252:     /// `merkleize`.

254:         self,

257:         let Self {
258:             staged,
259:             db,
260:             metadata,
261:         } = self;
262:         let (range, values, staged) = {
263:             let guard = db.read().await;
264:             staged.expand(keys, &guard).await?
265:         };
266:         Ok((
267:             range,
268:             values,
269:             Self {
270:                 staged,
271:                 db,
272:                 metadata,
273:             },
274:         ))
275:     }

278: /// Staged merkleize for the `any` unordered update kind.
279: impl<F, E, C, I, H, K, V, S> AnyStaged<F, E, C, I, H, unordered::Update<K, V>, S>
280: where
281:     F: Family,

289:     Operation<F, unordered::Update<K, V>>: Codec,

291:     /// Record updates for staged reads and upserts for unread keys, then merkleize.
292:     ///
293:     /// Consumes the staged handle and write vectors. Call [`expand`](AnyStaged::expand) before
294:     /// this method if more keys must be read into the staged index space.
295:     ///

302:     ///
303:     /// Panics if any update's `read_index` is out of the staged read range.
304:     pub async fn merkleize(

308:     ) -> Result<AnyMerkleized<F, E, C, I, H, unordered::Update<K, V>, S>, Error<F>> {
309:         let Self {
310:             staged,
311:             db,
312:             metadata,
313:         } = self;
314:         let inner = {
315:             let guard = db.read().await;
316:             staged.merkleize(updates, upserts, metadata, &guard).await?
317:         };
318:         Ok(AnyMerkleized { inner, db })
319:     }
320: }
