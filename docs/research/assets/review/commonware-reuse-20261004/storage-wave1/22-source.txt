93: /// Whether a completed sync journal's `bounds` cover `range`: retained data reaches back to
94: /// `range.start()` and ends exactly at `range.end()`.
95: pub(crate) fn journal_covers_range<F: Family>(
96:     bounds: std::ops::Range<u64>,
97:     range: &NonEmptyRange<Location<F>>,
98: ) -> bool {
99:     Location::new(bounds.start) <= range.start() && Location::new(bounds.end) == range.end()
100: }
101: 
102: /// Shared body for [`Database::local_pinned_nodes`] implementations backed by a persisted
103: /// [`full::Merkle`]. Reopens it from `config` under `context` and returns the pinned nodes at
104: /// `target.range.start()` if the persisted bounds cover the target and the root, computed with
105: /// `inactivity_floor`, matches `target.root`. Returns `Ok(None)` when the persisted state
106: /// cannot authenticate the target, including when an interrupted reset left required local
107: /// nodes unavailable.
108: pub(crate) async fn local_pinned_nodes<F, E, H, S>(
109:     context: E,
110:     config: full::Config<S>,
111:     target: &Target<F, H::Digest>,
112:     inactivity_floor: Location<F>,
113: ) -> Result<Option<Vec<H::Digest>>, crate::qmdb::Error<F>>
114: where
115:     F: Family,
116:     E: Context,
117:     H: Hasher,
118:     S: Strategy,
119: {
120:     let hasher = crate::qmdb::hasher::<H>();
121: 
122:     // A crash can persist a node-journal reset before its replacement metadata.
123:     // Missing local pins then use a peer-authenticated boundary. Other errors still propagate.
124:     let merkle = match full::Merkle::<F, _, _, S>::init(context, &hasher, config).await {
125:         Ok(merkle) => merkle,
126:         Err(crate::merkle::Error::MissingNode(_)) => return Ok(None),
127:         Err(err) => return Err(err.into()),
128:     };
129:     let bounds = merkle.bounds();
130:     if bounds.start > target.range.start() || bounds.end != target.range.end() {
131:         return Ok(None);
132:     }
133: 
134:     let inactive_peaks = F::inactive_peaks(target.range.end(), inactivity_floor);
135:     if merkle.root(&hasher, inactive_peaks)? != target.root {
136:         return Ok(None);
137:     }
138: 
139:     merkle
140:         .pinned_nodes_at(target.range.start())
141:         .await
142:         .map(Some)
143:         .map_err(Into::into)
144: }
