86:     async fn setup_network_and_peers(
87:         context: &deterministic::Context,
88:         peer_seeds: &[u64],
89:     ) -> (
90:         Oracle<PublicKey, deterministic::Context>,
91:         Vec<PrivateKey>,
92:         Vec<PublicKey>,
93:         Vec<(
94:             (
95:                 Sender<PublicKey, deterministic::Context>,
96:                 Receiver<PublicKey>,
97:             ),
98:             (
99:                 Sender<PublicKey, deterministic::Context>,
100:                 Receiver<PublicKey>,
101:             ),
102:         )>,
103:     ) {
104:         let (network, oracle) = Network::new(
105:             context.child("network"),
106:             commonware_p2p::simulated::Config {
107:                 max_size: 1024 * 1024,
108:                 max_peers_per_set: NZUsize!(peer_seeds.len()),
109:                 disconnect_on_block: true,
110:                 tracked_peer_sets: NZUsize!(1),
111:             },
112:         );
113:         network.start();
114: 
115:         let schemes: Vec<PrivateKey> = peer_seeds
116:             .iter()
117:             .map(|seed| PrivateKey::from_seed(*seed))
118:             .collect();
119:         let peers: Vec<PublicKey> = schemes.iter().map(|s| s.public_key()).collect();
120: 
121:         let mut connections = Vec::new();
122:         for peer in &peers {
123:             let control = oracle.control(peer.clone());
124:             let (sender1, receiver1) = control.register(0, TEST_QUOTA).await.unwrap();
125:             let (sender2, receiver2) = control.register(1, TEST_QUOTA).await.unwrap();
126:             connections.push(((sender1, receiver1), (sender2, receiver2)));
127:         }
128:         oracle
129:             .manager()
130:             .track(0, Set::from_iter_dedup(peers.clone()));
131: 
132:         (oracle, schemes, peers, connections)
133:     }

842:     fn spawn_engines_with_handles(
843:         engine_context: deterministic::Context,
844:         oracle: &Oracle<PublicKey, deterministic::Context>,
845:         schemes: Vec<PrivateKey>,
846:         connections: Vec<(
847:             (
848:                 Sender<PublicKey, deterministic::Context>,
849:                 Receiver<PublicKey>,
850:             ),
851:             (
852:                 Sender<PublicKey, deterministic::Context>,
853:                 Receiver<PublicKey>,
854:             ),
855:         )>,
856:     ) -> (
857:         Vec<Mailbox<PublicKey, Request>>,
858:         Vec<commonware_runtime::Handle<()>>,
859:     ) {
860:         let mut mailboxes = Vec::new();
861:         let mut handles = Vec::new();
862: 
863:         for (idx, (scheme, conn)) in schemes.into_iter().zip(connections).enumerate() {
864:             let ctx = engine_context.child("peer").with_attribute("index", idx);
865:             let (mon, _) = MockMonitor::new();
866:             let (handler, _) = MockHandler::new(true);
867:             let (engine, mailbox) = Engine::new(
868:                 ctx,
869:                 Config {
870:                     blocker: oracle.control(scheme.public_key()),
871:                     monitor: mon,
872:                     handler,
873:                     mailbox_size: MAILBOX_SIZE,
874:                     priority_request: false,
875:                     request_codec: (),
876:                     priority_response: false,
877:                     response_codec: (),
878:                 },
879:             );
880:             handles.push(engine.start(conn.0, conn.1));
881:             mailboxes.push(mailbox);
882:         }
883: 
884:         (mailboxes, handles)
885:     }
