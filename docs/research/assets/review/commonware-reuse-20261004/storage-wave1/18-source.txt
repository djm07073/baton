19: /// A request for operations from a source's log.
20: pub enum Request<F: Family> {
21:     /// Fetch the operations in `[start, start + max_ops)`.
22:     Operations {
23:         /// Prove against the root the database had at this size.
24:         size: Location<F>,
25:         /// First operation to return.
26:         start: Location<F>,
27:         /// Maximum number of operations to return.
28:         max_ops: NonZeroU64,

30:     /// Fetch the single operation at `start` plus the pinned nodes at `start`, the lowest
31:     /// location the client will retain. The proof in the response authenticates the pinned nodes,
32:     /// so there is no way to request them on their own.

34:         /// Prove against the root the database had at this size.

38:     },

481: where
482:     F: Family,
483:     E: Context,
484:     C: Contiguous<Item: EncodeShared + Floored<F>>,
485:     H: Hasher,
486:     S: Strategy,

493:     #[allow(clippy::type_complexity)]
494:     #[tracing::instrument(
495:         name = "qmdb.sync.serve",
496:         level = "info",
497:         skip_all,
498:         fields(
499:             size = *request.size(),
500:             start = *request.start(),
501:             max_ops = request.max_ops().get(),
502:         ),
503:     )]
504:     async fn serve(
505:         &self,
506:         request: Request<F>,
507:     ) -> Result<(Response<F, C::Item, H::Digest>, FeedbackTx), qmdb::Error<F>> {
508:         // Reject before the floor lookup so the error carries the requested size and the
509:         // floor read never touches out-of-range locations.
510:         if request.size() > self.size() {
511:             return Err(crate::merkle::Error::RangeOutOfBounds(request.size()).into());
512:         }
513:         let inactive_peaks = qmdb::inactive_peaks_at::<F, _>(self, request.size()).await?;
514:         let response = match request {
515:             Request::Operations {
516:                 size,
517:                 start,
518:                 max_ops,
519:             } => {
520:                 let (proof, operations) = self
521:                     .historical_proof(size, start, max_ops, inactive_peaks)
522:                     .await?;
523:                 Response::Operations { proof, operations }
524:             }
525:             Request::Boundary { size, start } => {
526:                 let (proof, mut operations) = self
527:                     .historical_proof(size, start, NonZeroU64::MIN, inactive_peaks)
528:                     .await?;
529:                 let op = operations
530:                     .pop()
531:                     .ok_or(crate::merkle::Error::RangeOutOfBounds(start))?;
532:                 let pinned_nodes = self.merkle.pinned_nodes_at(start).await?;
533:                 Response::Boundary {
534:                     proof,
535:                     op,
536:                     pinned_nodes,
537:                 }
538:             }
539:         };
540:         Ok((response, None))
541:     }
542: }
